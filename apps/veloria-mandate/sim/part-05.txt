export function advanceTurn(s){if(s.event||needsDispatch(s)||s.ending)return{ok:false,reason:s.event?'Сначала решите кризис':needsDispatch(s)?'Сначала отправьте квартальную депешу':'Партия окончена'};const r=rngFromSeed(`${s.seed}:${s.turn}:world`);delayedTick(s);economyTick(s,r);aiTick(s,r);imperialTick(s,r);s.turn++;s.month++;if(s.month>12){s.month=1;s.year++}const candidates=incidentCandidates(s);const crisisChance=.22+s.settings.crises/220+s.crisisHeat/310;if(candidates.length&&r()<crisisChance){const id=weighted(r,candidates);s.event=eventFor(id,s,r)}s.reports=worldReport(r,s);s.news=s.news.slice(0,9);normalize(s);endCheck(s);return{ok:true}}
