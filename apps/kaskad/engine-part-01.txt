function uid(prefix = 'id') {
  return `${prefix}-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 7)}`;
}

function clamp(value, min, max) {
  return Math.max(min, Math.min(max, value));
}

function numberOr(value, fallback) {
  const parsed = Number(value);
  return Number.isFinite(parsed) ? parsed : fallback;
}

function deepClone(value) {
  return JSON.parse(JSON.stringify(value));
}

function escapeHtml(value) {
  return String(value)
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#39;');
}

function downloadBlob(blob, filename) {
  const url = URL.createObjectURL(blob);
  const link = document.createElement('a');
  link.href = url;
  link.download = filename;
  document.body.append(link);
  link.click();
  link.remove();
  setTimeout(() => URL.revokeObjectURL(url), 1500);
}

function safeFilename(name) {
  return String(name || 'kaskad-scene')
    .trim()
    .toLowerCase()
    .replace(/[^a-zа-яё0-9]+/gi, '-')
    .replace(/^-+|-+$/g, '') || 'kaskad-scene';
}

function hexToRgb(hex) {
  const cleaned = String(hex).replace('#', '');
  const full = cleaned.length === 3 ? cleaned.split('').map((char) => char + char).join('') : cleaned;
  const number = Number.parseInt(full, 16);
  if (!Number.isFinite(number)) return { r: 255, g: 255, b: 255 };
  return { r: (number >> 16) & 255, g: (number >> 8) & 255, b: number & 255 };
}

function hexToHue(hex) {
  const { r, g, b } = hexToRgb(hex);
  const rn = r / 255;
  const gn = g / 255;
  const bn = b / 255;
  const max = Math.max(rn, gn, bn);
  const min = Math.min(rn, gn, bn);
  const delta = max - min;
  if (delta === 0) return 0;
  let hue;
  if (max === rn) hue = ((gn - bn) / delta) % 6;
  else if (max === gn) hue = (bn - rn) / delta + 2;
  else hue = (rn - gn) / delta + 4;
  return ((hue * 60) + 360) % 360;
}

function hueDistance(a, b) {
  const diff = Math.abs(a - b) % 360;
  return Math.min(diff, 360 - diff) / 180;
}

  return {
    id: typeof raw?.id === 'string' ? raw.id : uid('species'),

    color: /^#[0-9a-f]{6}$/i.test(raw?.color || '') ? raw.color : PALETTE[index % PALETTE.length],

}

function normalizeScene(raw) {
  const fallback = starterClone();
  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return fallback;
  const species = Array.isArray(raw.species) ? raw.species.slice(0, MAX_SPECIES).map(normalizeSpecies) : fallback.species;
  if (species.length === 0) species.push(normalizeSpecies({}, 0));
  const scene = {
    schema: 1,
    id: typeof raw.id === 'string' ? raw.id : uid('scene'),
    name: typeof raw.name === 'string' && raw.name.trim() ? raw.name.trim().slice(0, 40) : fallback.name,
    boundary: raw.boundary === 'box' ? 'box' : 'circle',
    background: /^#[0-9a-f]{6}$/i.test(raw.background || '') ? raw.background : fallback.background,
    trails: clamp(numberOr(raw.trails, 0), 0, 0.92),
    collisions: raw.collisions !== false,
    maxParticles: clamp(Math.round(Number(raw.maxParticles) || 180), 20, 500),
    speed: SPEED_STEPS.includes(Number(raw.speed)) ? Number(raw.speed) : 1,
    sound: raw.sound === true,
    selectedSpecies: typeof raw.selectedSpecies === 'string' ? raw.selectedSpecies : species[0].id,
    species,
    fields: Array.isArray(raw.fields) ? raw.fields.slice(0, MAX_FIELDS).map(normalizeField) : [],
    rules: Array.isArray(raw.rules) ? raw.rules.slice(0, MAX_RULES).map(normalizeRule) : []
  };
  if (!scene.species.some((item) => item.id === scene.selectedSpecies)) scene.selectedSpecies = scene.species[0].id;
  const speciesIds = new Set(scene.species.map((item) => item.id));
  for (const rule of scene.rules) {
    if (rule.source !== 'any' && !speciesIds.has(rule.source)) rule.source = 'any';
    if (rule.target !== 'any' && !speciesIds.has(rule.target)) rule.target = 'any';
    if (rule.outputSpecies !== 'same' && !speciesIds.has(rule.outputSpecies)) rule.outputSpecies = 'same';
    if (rule.outputField !== 'any' && !scene.fields.some((field) => field.id === rule.outputField)) rule.outputField = 'any';
  }
  return scene;
}
