function ruleSentence(rule) {
  const trigger = TRIGGER_DEFS[rule.trigger].short;
  const source = getSpeciesLabel(rule.source);

}

function conditionThresholdConfig(condition) {
  if (condition === 'speedAbove') return { min: 0, max: 500, step: 5, suffix: ' px/с', defaultValue: 80 };
  if (condition === 'populationBelow') return { min: 1, max: 500, step: 1, suffix: '', defaultValue: 120 };
  if (condition === 'ageAbove') return { min: 0, max: 30, step: 0.5, suffix: ' с', defaultValue: 3 };
  if (condition === 'sameHue' || condition === 'oppositeHue') return { min: 0.02, max: 1, step: 0.02, suffix: '', defaultValue: condition === 'sameHue' ? 0.18 : 0.6 };
  return { min: 0, max: 1, step: 0.1, suffix: '', defaultValue: 0 };
}

function updateRuleEditorVisibility() {
  const form = document.querySelector('#rule-form');
  if (!form) return;
  const trigger = form.elements.trigger.value;
  const condition = form.elements.condition.value;
  const action = form.elements.action.value;
  const targetField = form.querySelector('[data-target-field]');
  const outputSpeciesField = form.querySelector('[data-output-species-field]');
  const outputFieldField = form.querySelector('[data-output-field-field]');
  const thresholdWrap = form.querySelector('[data-threshold-wrap]');
  targetField.hidden = trigger !== 'collision';
  outputSpeciesField.hidden = !['spawn', 'split', 'transform', 'burst'].includes(action);
  outputFieldField.hidden = action !== 'toggleField';
  thresholdWrap.hidden = condition === 'always';
  const intervalInput = form.elements.interval.closest('.form-field');
  intervalInput.hidden = trigger !== 'interval';

  const thresholdConfig = conditionThresholdConfig(condition);
  const thresholdInput = form.elements.threshold;
  if (thresholdInput) {
    thresholdInput.min = thresholdConfig.min;
    thresholdInput.max = thresholdConfig.max;
    thresholdInput.step = thresholdConfig.step;
    thresholdInput.dataset.suffix = thresholdConfig.suffix;
    if (Number(thresholdInput.value) < thresholdConfig.min || Number(thresholdInput.value) > thresholdConfig.max) thresholdInput.value = thresholdConfig.defaultValue;
    updateRangeOutput(thresholdInput);
  }
  updateRulePreview();
}

function updateRulePreview() {
  const form = document.querySelector('#rule-form');
  const preview = document.querySelector('#rule-preview');
  if (!form || !preview) return;
  const trigger = TRIGGER_DEFS[form.elements.trigger.value]?.label || 'Событие';
  const source
