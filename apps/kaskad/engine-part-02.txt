function randomPointInBoundary(radius = 0) {
  if (scene.boundary === 'circle') {
    const bounds = getBoundary();
    const angle = Math.random() * Math.PI * 2;
    const distance = Math.sqrt(Math.random()) * Math.max(1, bounds.radius - radius - 3);
    return { x: bounds.cx + Math.cos(angle) * distance, y: bounds.cy + Math.sin(angle) * distance };
  }
  const margin = 18 + radius;
  return {
    x: margin + Math.random() * Math.max(1, viewport.width - margin * 2),
    y: margin + Math.random() * Math.max(1, viewport.height - margin * 2)
  };
}

function getBoundary() {
  const margin = Math.max(18, Math.min(viewport.width, viewport.height) * 0.045);
  if (scene.boundary === 'circle') {
    return {
      type: 'circle',
      cx: viewport.width / 2,
      cy: viewport.height / 2,
      radius: Math.max(20, Math.min(viewport.width, viewport.height) / 2 - margin)
    };
  }
  return { type: 'box', left: margin, top: margin, right: viewport.width - margin, bottom: viewport.height - margin };
}

function spawnParticle(speciesId, options = {}) {
  if (particles.length >= scene.maxParticles) return null;
  const species = getSpecies(speciesId);
  if (!species) return null;
  const point = Number.isFinite(options.x) && Number.isFinite(options.y)
    ? { x: options.x, y: options.y }
    : randomPointInBoundary(species.radius);
  const angle = Number.isFinite(options.angle) ? options.angle : Math.random() * Math.PI * 2;
  const speed = Number.isFinite(options.speed) ? options.speed : species.speed * (0.55 + Math.random() * 0.9);
  const particle = {
    id: nextParticleId++,
    speciesId: species.id,
    x: point.x,
    y: point.y,
    vx: Number.isFinite(options.vx) ? options.vx : Math.cos(angle) * speed,
    vy: Number.isFinite(options.vy) ? options.vy : Math.sin(angle) * speed,
    r: clamp(Number(options.radius) || species.radius, 2, 50),
    age: 0,
    tint: options.tint || null,
    dead: false
  };
  constrainParticle(particle, false);
  particles.push(particle);
  return particle;
}

function resetSimulation() {
  particles = [];
  effects = [];
  currentContacts.clear();
  ruleLastTriggered.clear();
  intervalSchedule.clear();
  simTime = 0;
  nextParticleId = 1;
  for (const species of scene.species) {
    const count = Math.min(species.count, scene.maxParticles - particles.length);
    for (let index = 0; index < count; index += 1) spawnParticle(species.id);
  }
  lastCanvasFill = '';
  updateHud(true);
}

function constrainParticle(particle, bounce = true) {
  const species = getSpecies(particle.speciesId);
  const restitution = species?.bounce ?? 0.95;
  const bounds = getBoundary();
  let hit = false;
  if (bounds.type === 'circle') {
    const dx = particle.x - bounds.cx;
    const dy = particle.y - bounds.cy;
    const distance = Math.hypot(dx, dy) || 0.0001;
    const maxDistance = Math.max(1, bounds.radius - particle.r);
    if (distance > maxDistance) {
      const nx = dx / distance;
      const ny = dy / distance;
      particle.x = bounds.cx + nx * maxDistance;
      particle.y = bounds.cy + ny * maxDistance;
      const normalVelocity = particle.vx * nx + particle.vy * ny;
      if (bounce && normalVelocity > 0) {
        particle.vx -= (1 + restitution) * normalVelocity * nx;
        particle.vy -= (1 + restitution) * normalVelocity * ny;
        hit = true;
      }
    }
  } else {
    if (particle.x - particle.r < bounds.left) {
      particle.x = bounds.left + particle.r;
      if (bounce && particle.vx < 0) { particle.vx = Math.abs(particle.vx) * restitution; hit = true; }
    } else if (particle.x + particle.r > bounds.right) {
      particle.x = bounds.right - particle.r;
      if (bounce && particle.vx > 0) { particle.vx = -Math.abs(particle.vx) * restitution; hit = true; }
    }
    if (particle.y - particle.r < bounds.top) {
      particle.y = bounds.top + particle.r;
      if (bounce && particle.vy < 0) { particle.vy = Math.abs(particle.vy) * restitution; hit = true; }
    } else if (particle.y + particle.r > bounds.bottom) {
      particle.y = bounds.bottom - particle.r;
      if (bounce && particle.vy > 0) { particle.vy = -Math.abs(particle.vy) * restitution; hit = true; }
    }
  }
  if (hit) {
    const speed = Math.hypot(particle.vx, particle.vy);
    emitEvent('wall', { source: particle, target: null, x: particle.x, y: particle.y, speed });
    addEffect(particle.x, particle.y, particle.tint || species.color, Math.min(28, 6 + speed * 0.025));
    playCollisionSound(particle, speed);
  }
}

function addEffect(x, y, color, size = 12) {
  if (effects.length > 90) effects.shift();
  effects.push({ x, y, color, size, age: 0, life: 0.38 });
}

function applyFields(dt) {
  const activeFields = scene.fields.filter((field) => field.enabled);
  for (const particle of particles) {
    if (particle.dead) continue;
    const species = getSpecies(particle.speciesId);
    const inverseMass = 1 / Math.max(0.1, species.mass);
    for (const field of activeFields) {
      if (field.type === 'hue') continue;
      const anchorX = viewport.width * field.x;
      const anchorY = viewport.height * field.y;
      const dx = anchorX - particle.x;
      const dy = anchorY - particle.y;
      const distance = Math.hypot(dx, dy) || 0.0001;
      const normalizedRadius = Math.max(30, Math.min(viewport.width, viewport.height) * field.radius);
      const influence = clamp(1 - distance / normalizedRadius, 0, 1);
      if (field.type === 'gravity') {
        particle.vy += field.strength * 165 * dt * inverseMass;
      } else if (field.type === 'attractor' && influence > 0) {
        const force = field.strength * 360 * (0.18 + influence * influence);
        particle.vx += dx / distance * force * dt * inverseMass;
        particle.vy += dy / distance * force * dt * inverseMass;
      } else if (field.type === 'vortex' && influence > 0) {
        const force = field.strength * 330 * (0.2 + influence);
        particle.vx += -dy / distance * force * dt * inverseMass;
        particle.vy += dx / distance * force * dt * inverseMass;
      } else if (field.type === 'wind') {
        const angle = field.frequency * Math.PI * 0.5;
        particle.vx += Math.cos(angle) * field.strength * 130 * dt * inverseMass;
        particle.vy += Math.sin(angle) * field.strength * 130 * dt * inverseMass;
      } else if (field.type === 'turbulence') {
        const phase = simTime * field.frequency + particle.id * 1.618;
        particle.vx += Math.cos(phase * 2.31 + Math.sin(particle.y * 0.018)) * field.strength * 170 * dt;
        particle.vy += Math.sin(phase * 1.77 + Math.cos(particle.x * 0.016)) * field.strength * 170 * dt;
      }
    }
  }

  const hueFields = activeFields.filter((field) => field.type === 'hue');
  if (hueFields.length === 0 || particles.length > 320) return;
  const combinedStrength = hueFields.reduce((sum, field) => sum + field.strength, 0);
  const maxDistance = 230;
  for (let i = 0; i < particles.length; i += 1) {
    const a = particles[i];
    if (a.dead) continue;
    const speciesA = getSpecies(a.speciesId);
    const hueA = hexToHue(a.tint || speciesA.color);
    for (let j = i + 1; j < particles.length; j += 1) {
      const b = particles[j];
      if (b.dead) continue;
      const dx = b.x - a.x;
      const dy = b.y - a.y;
      const distanceSq = dx * dx + dy * dy;
      if (distanceSq > maxDistance * maxDistance || distanceSq < 4) continue;
      const distance = Math.sqrt(distanceSq);
      const speciesB = getSpecies(b.speciesId);
      const hueB = hexToHue(b.tint || speciesB.color);
      const colorPhase = Math.cos(hueDistance(hueA, hueB) * Math.PI);
      const falloff = 1 - distance / maxDistance;
      const force = colorPhase * combinedStrength * 125 * falloff * falloff;
      const nx = dx / distance;
      const ny = dy / distance;
      a.vx -= nx * force * dt / Math.max(0.2, speciesA.mass);
      a.vy -= ny * force * dt / Math.max(0.2, speciesA.mass);
      b.vx += nx * force * dt / Math.max(0.2, speciesB.mass);
      b.vy += ny * force * dt / Math.max(0.2, speciesB.mass);
    }
  }
}

function resolveParticleCollisions() {
  if (!scene.collisions) {
    currentContacts.clear();
    return;
  }
  const nextContacts = new Set();
  for (let i = 0; i < particles.length; i += 1) {
    const a = particles[i];
    if (a.dead) continue;
    const speciesA = getSpecies(a.speciesId);
    for (let j = i + 1; j < particles.length; j += 1) {
      const b = particles[j];
      if (b.dead) continue;
      const dx = b.x - a.x;
      const dy = b.y - a.y;
      const minDistance = a.r + b.r;
      const distanceSq = dx * dx + dy * dy;
      if (distanceSq >= minDistance * minDistance || distanceSq === 0) continue;
      const distance = Math.sqrt(distanceSq);
      const nx = dx / distance;
      const ny = dy / distance;
      const overlap = minDistance - distance;
      const speciesB = getSpecies(b.speciesId);
      const massA = Math.max(0.1, speciesA.mass * (a.r / speciesA.radius) ** 2);
      const massB = Math.max(0.1, speciesB.mass * (b.r / speciesB.radius) ** 2);
      const totalMass = massA + massB;
      a.x -= nx * overlap * (massB / totalMass);
      a.y -= ny * overlap * (massB / totalMass);
      b.x += nx * overlap * (massA / totalMass);
      b.y += ny * overlap * (massA / totalMass);

      const relativeVelocity = (b.vx - a.vx) * nx + (b.vy - a.vy) * ny;
      if (relativeVelocity < 0) {
        const restitution = Math.min(speciesA.bounce, speciesB.bounce);
        const impulse = -(1 + restitution) * relativeVelocity / (1 / massA + 1 / massB);
        a.vx -= impulse * nx / massA;
        a.vy -= impulse * ny / massA;
        b.vx += impulse * nx / massB;
        b.vy += impulse * ny / massB;
      }

      const key = a.id < b.id ? `${a.id}:${b.id}` : `${b.id}:${a.id}`;
      nextContacts.add(key);
      if (!currentContacts.has(key)) {
        const impactSpeed = Math.abs(relativeVelocity);
        emitEvent('collision', { source: a, target: b, x: (a.x + b.x) / 2, y: (a.y + b.y) / 2, speed: impactSpeed });
        addEffect((a.x + b.x) / 2, (a.y + b.y) / 2, speciesA.color, Math.min(20, 5 + impactSpeed * 0.018));
        playCollisionSound(a, impactSpeed);
      }
    }
  }
  currentContacts = nextContacts;
}

function ruleMatchesSpecies(ruleValue, particle) {
  return ruleValue === 'any' || Boolean(particle && particle.speciesId === ruleValue);
}

function getEventHueDistance(event) {
  if (!event.source || !event.target) return 0;
  const sourceSpecies = getSpecies(event.source.speciesId);
  const targetSpecies = getSpecies(event.target.speciesId);
  return hueDistance(hexToHue(event.source.tint || sourceSpecies.color), hexToHue(event.target.tint || targetSpecies.color));
}

function ruleConditionPasses(rule, event) {
  const source = event.source;
  if (rule.condition === 'always') return true;
  if (rule.condition === 'speedAbove') return (event.speed || Math.hypot(source?.vx || 0, source?.vy || 0)) >= rule.threshold;
  if (rule.condition === 'populationBelow') return particles.length < rule.threshold;
  if (rule.condition === 'ageAbove') return (source?.age || 0) >= rule.threshold;
  if (rule.condition === 'sameHue') return Boolean(event.target) && getEventHueDistance(event) <= clamp(rule.threshold, 0.02, 1);
  if (rule.condition === 'oppositeHue') return Boolean(event.target) && getEventHueDistance(event) >= clamp(rule.threshold, 0.02, 1);
  return true;
}

function resolveRuleSource(rule, event) {
  if (rule.source === 'any') return event.source || event.target || particles[Math.floor(Math.random() * particles.length)] || null;
  if (event.source?.speciesId === rule.source) return event.source;
  if (event.target?.speciesId === rule.source) return event.target;
  if (event.trigger === 'interval') {
    const candidates = particles.filter((particle) => !particle.dead && particle.speciesId === rule.source);
    return candidates[Math.floor(Math.random() * candidates.length)] || null;
  }
  return null;
}

function resolveRuleTarget(rule, event, source) {
  if (rule.target === 'any') return event.target && event.target !== source ? event.target : null;
  if (event.target?.speciesId === rule.target && event.target !== source) return event.target;
  if (event.source?.speciesId === rule.target && event.source !== source) return event.source;
  return null;
}

function emitEvent(trigger, event) {
  eventCounter += 1;
  const now = simTime;
  for (const rule of scene.rules) {
    if (!rule.enabled || rule.trigger !== trigger) continue;
    const last = ruleLastTriggered.get(rule.id) ?? -Infinity;
    if (now - last < rule.cooldown) continue;
    const source = resolveRuleSource(rule, { ...event, trigger });
    if (rule.source !== 'any' && !source) continue;
    const target = resolveRuleTarget(rule, event, source);
    if (trigger === 'collision' && rule.target !== 'any' && !target) continue;
    const normalizedEvent = { ...event, trigger, source, target };
    if (!ruleConditionPasses(rule, normalizedEvent)) continue;
    if (Math.random() > rule.chance) continue;
    ruleLastTriggered.set(rule.id, now);
    applyRuleAction(rule, normalizedEvent);
  }
}

function outputSpeciesForRule(rule, source) {
  return rule.outputSpecies === 'same' ? source?.speciesId || scene.selectedSpecies : rule.outputSpecies;
}

function applyRuleAction(rule, event) {
  const source = event.source;
  const target = event.target;
  const x = Number.isFinite(event.x) ? event.x : source?.x ?? viewport.width / 2;
  const y = Number.isFinite(event.y) ? event.y : source?.y ?? viewport.height / 2;
