  showToast(existing ? 'Вид обновлён' : 'Вид добавлен');
}

function deleteCurrent(kind) {
  if (kind === 'species') {
    const saveButton = sheetFooter.querySelector('[data-save-species]');
    const id = saveButton?.dataset.saveSpecies;
    const item = scene.species.find((species) => species.id === id);
    if (!item) return;
    askConfirm({
      title: `Удалить «${item.name}»?`,
      message: 'Правила, которые ссылались на этот вид, будут переключены на «любой» или «тот же».',
      confirmLabel: 'Удалить вид',
      danger: true,
      onConfirm() {
        commitScene((draft) => {
          draft.species = draft.species.filter((species) => species.id !== id);
          if (draft.selectedSpecies === id) draft.selectedSpecies = draft.species[0].id;
          for (const rule of draft.rules) {
            if (rule.source === id) rule.source = 'any';
            if (rule.target === id) rule.target = 'any';
            if (rule.outputSpecies === id) rule.outputSpecies = 'same';
          }
        }, { restart: true });
        showToast('Вид удалён');
      }
    });
  } else if (kind === 'field') {
    const id = sheetFooter.querySelector('[data-save-field]')?.dataset.saveField;
    const item = scene.fields.find((field) => field.id === id);
    if (!item) return;
    askConfirm({
      title: `Удалить «${item.name}»?`,
      message: 'Правила переключения этого поля будут использовать первое доступное поле.',
      confirmLabel: 'Удалить поле',
      danger: true,
      onConfirm() {
        commitScene((draft) => {
          draft.fields = draft.fields.filter((field) => field.id !== id);
          for (const rule of draft.rules) if (rule.outputField === id) rule.outputField = 'any';
        });
        showToast('Поле удалено');
      }
    });
  } else if (kind === 'rule') {
    const id = sheetFooter.querySelector('[data-save-rule]')?.dataset.saveRule;
    const item = scene.rules.find((rule) => rule.id === id);
    if (!item) return;
    askConfirm({
      title: `Удалить «${item.name}»?`,
      message: 'Физика сцены останется, но это событие больше не будет запускать действие.',
      confirmLabel: 'Удалить правило',
      danger: true,
      onConfirm() {
        commitScene((draft) => { draft.rules = draft.rules.filter((rule) => rule.id !== id); });
        showToast('Правило удалено');
      }
    });
  }
}

function updateRangeOutput(input) {
  const form = input.closest('form');
  const output = form?.querySelector(`[data-output-for="${CSS.escape(input.name)}"]`);
  if (!output) return;
  const step = Number(input.step);
  const value = step < 0.01 ? Number(input.value).toFixed(3) : step < 1 ? Number(input.value).toFixed(2) : Math.round(Number(input.value));
  output.textContent = `${value}${input.dataset.suffix || ''}`;
}

function exportScene() {
  const payload = {
    schema: 'kaskad-scene',
    version: 1,
    appVersion: APP_VERSION,
    exportedAt: new Date().toISOString(),
    scene
  };
  downloadBlob(new Blob([JSON.stringify(payload, null, 2)], { type: 'application/json' }), `${safeFilename(scene.name)}.kaskad.json`);
  showToast('Сцена экспортирована');
}

function importSceneFile(file) {
  const reader = new FileReader();
  reader.onload = () => {
    try {
      const payload = JSON.parse(String(reader.result));
      if (payload?.schema !== 'kaskad-scene' || !payload.scene) throw new Error('Неверный формат KASKAD');
      const imported = normalizeScene(payload.scene);
      pushUndo();
      scene = imported;
      scheduleSave();
      updateSceneChrome();
      resetSimulation();
      renderPanel();
      closeSheet();
      showToast('Сцена импортирована');
    } catch (error) {
      showToast(error.message || 'Не удалось прочитать файл', 3200);
    }
  };
  reader.onerror = () => showToast('Не удалось прочитать файл', 3200);
  reader.readAsText(file);
}

function toggleRecording() {
  if (mediaRecorder?.state === 'recording') {
    mediaRecorder.stop();
    return;
  }
  if (!canvas.captureStream || typeof MediaRecorder === 'undefined') {
    showToast('Видео не поддерживается — сохраняю кадр');
    takeScreenshot();
    return;
  }
  try {
    const stream = canvas.captureStream(60);
    const candidates = ['video/webm;codecs=vp9', 'video/webm;codecs=vp8', 'video/webm'];
    const mimeType = candidates.find((type) => MediaRecorder.isTypeSupported?.(type)) || '';
    mediaRecorder = new MediaRecorder(stream, mimeType ? { mimeType, videoBitsPerSecond: 5_000_000 } : undefined);
    recordingChunks = [];
    mediaRecorder.ondataavailable = (event) => { if (event.data?.size) recordingChunks.push(event.data); };
    mediaRecorder.onstop = () => {
      const blob = new Blob(recordingChunks, { type: mediaRecorder.mimeType || 'video/webm' });
      if (blob.size > 0) downloadBlob(blob, `${safeFilename(scene.name)}-${Date.now()}.webm`);
      recordingBadge.hidden = true;
      recordButton.classList.remove('is-recording');
      recordButton.querySelector('span').textContent = 'REC';
      showToast(`Запись сохранена · ${Math.max(1, Math.round((performance.now() - recordingStartedAt) / 1000))} с`);
      mediaRecorder = null;
      recordingChunks = [];
    };
    mediaRecorder.start(250);
    recordingStartedAt = performance.now();
    recordingBadge.hidden = false;
    recordButton.classList.add('is-recording');
    recordButton.querySelector('span').textContent = 'STOP';
    showToast('Запись началась');
  } catch {
    showToast('Видео не запустилось — сохраняю кадр');
    takeScreenshot();
  }
}

function canvasPoint(event) {
  const rect = canvas.getBoundingClientRect();
  return { x: event.clientX - rect.left, y: event.clientY - rect.top };
}

function spawnFromGesture(point, velocity = null) {
  const species = getSpecies(scene.selectedSpecies);
  const particle = spawnParticle(species.id, {
    x: point.x,
    y: point.y,
    vx: velocity?.vx,
    vy: velocity?.vy,
    speed: velocity ? Math.hypot(velocity.vx, velocity.vy) : species.speed
  });
  if (particle) {
    emitEvent('tap', { source: particle, target: null, x: point.x, y: point.y, speed: Math.hypot(particle.vx, particle.vy) });
    addEffect(point.x, point.y, species.color, 15);
  } else showToast(`Достигнут лимит: ${scene.maxParticles}`);
}

canvas.addEventListener('pointerdown', (event) => {
  if (event.button !== 0 && event.pointerType === 'mouse') return;
  event.preventDefault();
  canvas.setPointerCapture(event.pointerId);
  const point = canvasPoint(event);
  pointerState = { id: event.pointerId, start: point, last: point, lastAt: performance.now(), moved: false, emittedAt: 0 };
  spawnFromGesture(point);
  haptic(5);
});
