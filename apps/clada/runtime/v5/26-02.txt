    const middleX = (point.x + neighbor.x) / 2;
    const middleY = (point.y + neighbor.y) / 2;
    const normalX = neighbor.x - point.x;
    const normalY = neighbor.y - point.y;
    const inside = (vertex) => (vertex[0] - middleX) * normalX + (vertex[1] - middleY) * normalY <= 1e-9;
    const intersection = (start, end) => {
      const startValue = (start[0] - middleX) * normalX + (start[1] - middleY) * normalY;
      const endValue = (end[0] - middleX) * normalX + (end[1] - middleY) * normalY;
      const denominator = startValue - endValue;
      const t = Math.abs(denominator) < 1e-12 ? 0 : startValue / denominator;
      return [lerp(start[0], end[0], t), lerp(start[1], end[1], t)];
    };
    const output = [];
    for (let index = 0; index < polygon.length; index += 1) {
      const current = polygon[index];
      const previous = polygon[(index + polygon.length - 1) % polygon.length];
      const currentInside = inside(current);
      const previousInside = inside(previous);
      if (currentInside && !previousInside) output.push(intersection(previous, current));
      if (currentInside) output.push(current);
      else if (previousInside) output.push(intersection(previous, current));
    }
    return output;
  }

  function voronoiPolygon(points, neighbors, index) {
    let polygon = [[.012, .018], [.988, .018], [.988, .982], [.012, .982]];
    for (const neighborIndex of neighbors[index]) polygon = clipHalfPlane(polygon, points[index], points[neighborIndex]);
    return polygon.map(([x, y]) => [round(clamp(x, 0, 1), 5), round(clamp(y, 0, 1), 5)]);
  }

  function nearestPatch(community, point) {
    return [...(community.patches || [])].sort((a, b) => ((a.x - point.x) ** 2 + (a.y - point.y) ** 2) - ((b.x - point.x) ** 2 + (b.y - point.y) ** 2))[0] || null;
  }

  function baseStateForPatch(patch) {
    const water = Boolean(patch?.water);
    const forest = patch?.biome === 'forest';
    const wet = patch?.biome === 'wetland';
    return {
      canopy: forest ? .68 : water ? .02 : .28 + clamp(patch?.moisture || .4) * .18,
      openness: forest ? .18 : water ? .04 : .52,
      soil: water ? .05 : .3 + clamp(patch?.moisture || .4) * .2,
      wetland: wet ? .72 : water ? .2 : clamp((patch?.moisture || .3) - .45) * .48,
      reef: water ? .08 : 0,
      turbidity: water ? .32 : .04,
      fire: 0,
      organic: water ? .18 : .22,
      channels: wet ? .36 : .08,
      nutrients: .28,
      trails: .04
    };
  }

  function buildCells(community, trace) {
    const points = relaxPoints(initialPoints(trace));
    const triangles = delaunay(points);
    const neighbors = buildNeighbors(points, triangles);
    trace.cells = points.map((point, index) => {
      const patch = nearestPatch(community, point);
      const state = baseStateForPatch(patch);
      return {
        id: `c${index}`,
        x: round(point.x, 5), y: round(point.y, 5),
        polygon: voronoiPolygon(points, neighbors, index),
        neighbors: neighbors[index].map((neighbor) => `c${neighbor}`),
        patchId: patch?.id ?? null,
        baseBiome: patch?.biome || 'grassland',
        water: Boolean(patch?.water),
        state,
        dominant: dominantEffect(state),
        dominantStrength: 0,
        constructionStrength: 0,
        engineerSpeciesIds: [],
        engineerScores: {},
        engineerEffects: {},
        history: []
      };
    });
    trace.triangles = triangles.map((triangle) => [triangle.a, triangle.b, triangle.c]);
    trace.geometryVersion = 1;
  }

  let connectivityBridgeInstalled = false;

  function installConnectivityBridge() {
    if (connectivityBridgeInstalled) return;
    const planet = globalThis.CladaLivingPlanetCore;
    if (!planet || typeof planet.connectivityModifier !== 'function') return;
    const legacy = planet.connectivityModifier.bind(planet);
    planet.connectivityModifier = function lifeTraceConnectivityModifier(community, source, target, deme) {
      const value = legacy(community, source, target, deme);
      if (!Number.isFinite(value)) return value;
      const sourceMultiplier = mobilityMultiplier(source, deme?.guild);
      const targetMultiplier = mobilityMultiplier(target, deme?.guild);
      return clamp(value * Math.sqrt(sourceMultiplier * targetMultiplier), .002, 1.45);
    };
    connectivityBridgeInstalled = true;
  }

  function ensureCommunity(community) {
    installConnectivityBridge();
    const trace = traceState(community);
    if (trace.cells.length !== CELL_COUNT || trace.geometryVersion !== 1) buildCells(community, trace);
    const patchIds = new Set((community.patches || []).map((patch) => String(patch.id)));
    for (const cell of trace.cells) {
      if (!patchIds.has(String(cell.patchId))) cell.patchId = nearestPatch(community, cell)?.id ?? null;
      cell.state ||= baseStateForPatch((community.patches || []).find((patch) => String(patch.id) === String(cell.patchId)));
      cell.history ||= [];
      cell.engineerSpeciesIds ||= [];
      cell.engineerScores ||= {};
      cell.engineerEffects ||= {};
      cell.constructionStrength = clamp(cell.constructionStrength ?? cell.dominantStrength ?? 0);
    }
    return trace;
  }

  function dominantEffect(state) {
    const weighted = {
      canopy: state.canopy,
      openness: state.openness,
      soil: state.soil * .94,
      wetland: state.wetland * 1.12,
      reef: state.reef * 1.16,
      turbidity: state.turbidity * .82,
      fire: state.fire * 1.25,
      organic: state.organic * .88,
      channels: state.channels * .92,
      nutrients: state.nutrients * .82,
      trails: state.trails * .9
    };
    return Object.entries(weighted).sort((a, b) => b[1] - a[1])[0]?.[0] || 'soil';
  }
