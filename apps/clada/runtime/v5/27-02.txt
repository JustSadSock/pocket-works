  ctx.fillText(`${metrics.engineeredCells}/${trace.cells.length} ячеек изменены · болота ${metrics.wetlands} · рифы ${metrics.reefs}`, 20, 46);
  ctx.fillStyle = 'rgba(39,55,45,.64)'; ctx.font = '9px system-ui'; ctx.textAlign = 'right';
  ctx.fillText('коснись ячейки для причинной цепочки', cssWidth - 12, cssHeight - 12);
}

function lifeTraceDrawWorldMarks() {
  if (state.fossilIndex !== null || state.view !== 'world') return;
  const trace = lifeTraceEnsureState();
  if (!trace) return;
  const strongest = [...trace.cells].filter((cell) => cell.dominantStrength >= .58).sort((a, b) => b.dominantStrength - a.dominantStrength).slice(0, 18);
  ctx.save();
  ctx.globalAlpha = .32;
  ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
  for (const cell of strongest) {
    const x = cell.x * cssWidth, y = cell.y * cssHeight;
    ctx.strokeStyle = LIFE_TRACE_COLORS[cell.dominant] || '#5d6f60';
    ctx.fillStyle = LIFE_TRACE_COLORS[cell.dominant] || '#5d6f60';
    ctx.lineWidth = .8;
    if (cell.dominant === 'trails' || cell.dominant === 'channels') {
      ctx.setLineDash([3, 3]);
      ctx.beginPath(); ctx.moveTo(x - 8, y + 4); ctx.quadraticCurveTo(x, y - 7, x + 9, y + 2); ctx.stroke(); ctx.setLineDash([]);
    } else if (cell.dominant === 'fire') {
      ctx.beginPath(); ctx.arc(x, y, 7, 0, Math.PI * 2); ctx.stroke();
      ctx.beginPath(); ctx.moveTo(x - 4, y - 4); ctx.lineTo(x + 4, y + 4); ctx.moveTo(x + 4, y - 4); ctx.lineTo(x - 4, y + 4); ctx.stroke();
    } else {
      ctx.font = '700 11px system-ui'; ctx.fillText(LIFE_TRACE_GLYPHS[cell.dominant] || '·', x, y);
    }
  }
  ctx.restore();
}

drawWorld = function lifeTraceDrawWorld() {
  lifeTraceLegacyDrawWorld();
  lifeTraceDrawWorldMarks();
};

function lifeTraceEffectRows(report) {
  return Object.entries(report.state).sort((a, b) => b[1] - a[1]).slice(0, 8).map(([key, value]) => `<div class="life-trace-meter"><span><b>${safeText(LIFE_TRACE.EFFECT_LABELS[key] || key)}</b><small>${safeText(key === report.dominant ? 'главный устойчивый след' : 'сопутствующее изменение')}</small></span><output>${lifeTracePercent(value)}%</output><i style="--value:${lifeTracePercent(value)}%;--trace:${safeText(LIFE_TRACE_COLORS[key] || '#667668')}"></i></div>`).join('');
}

function lifeTraceHierarchyHtml(node, depth = 0) {
  if (!node) return '';
  const children = (node.children || []).map((child) => lifeTraceHierarchyHtml(child, depth + 1)).join('');
  const label = node.speciesId ? lifeTraceSpeciesName(node.speciesId) : node.label;
  return `<li style="--depth:${depth}"><span><i></i><b>${safeText(label)}</b><output>${lifeTracePercent(node.value || 0)}%</output></span>${children ? `<ul>${children}</ul>` : ''}</li>`;
}

function lifeTraceOpenCell(cellId) {
  const report = LIFE_TRACE.cellReport(state.metacommunity, cellId);
  if (!report) return;
  lifeTraceSelectedCellId = cellId;
  const engineers = report.engineerSpeciesIds.map((id) => `<button type="button" data-field-species="${id}"><span><b>${safeText(lifeTraceSpeciesName(id))}</b><small>вклад в изменение этой ячейки</small></span><output>${lifeTracePercent(report.engineerScores[id] || report.engineerScores[String(id)] || 0)}%</output></button>`).join('');
  const history = report.history.slice().reverse().map((entry) => `<div><b>G${entry.g}</b><span>${safeText(LIFE_TRACE.EFFECT_LABELS[entry.effect] || entry.effect)} · ${safeText(lifeTraceBiomeName(entry.biome))}</span><output>${lifeTracePercent(entry.strength)}%</output></div>`).join('');
  openSheet('СЛЕД ЖИЗНИ', `Ячейка ${report.id.slice(1)}`, `<div class="trace-cell-hero"><div><small>УЧАСТОК ${safeText(report.patchId)} · ${report.water ? 'ВОДА' : 'СУША'}</small><b>${safeText(report.dominantLabel)}</b><p>${safeText(lifeTraceBiomeName(report.biome))}</p></div><output>${lifeTracePercent(report.dominantStrength)}%</output></div><section class="field-section"><header><b>ЧТО ИЗМЕНИЛОСЬ</b><span>накопленный след, а не разовое действие</span></header><div class="life-trace-meters">${lifeTraceEffectRows(report)}</div></section><section class="field-section"><header><b>ПРИЧИННАЯ ЦЕПОЧКА</b><span>иерархия влияний и инженеров</span></header><ul class="trace-hierarchy">${lifeTraceHierarchyHtml(report.hierarchy)}</ul></section><section class="field-section"><header><b>ИНЖЕНЕРЫ ЭКОСИСТЕМЫ</b><span>${report.engineerSpeciesIds.length} активных линий</span></header><div class="trace-engineers">${engineers || '<div class="empty-mini">След пока поддерживается самой средой.</div>'}</div></section><section class="field-section"><header><b>ИСТОРИЯ ЯЧЕЙКИ</b><span>последние устойчивые состояния</span></header><div class="trace-history">${history || '<div class="empty-mini">История ещё не накоплена.</div>'}</div></section>`);
}
