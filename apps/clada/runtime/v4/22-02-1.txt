const OBSERVATION_APP_VERSION = '4.4.0';
const OBSERVATION = globalThis.CladaObservationCore;
const observationLegacySaveState = saveState;
const observationLegacySimulateStep = simulateStep;
const observationLegacyMetaRecordHistory = metaRecordHistory;
const observationLegacyBuildWorld = buildWorld;
const observationLegacyMigrateLivingState = migrateLivingState;
let observationSaveHandle = null;
let observationSavePending = false;

function observationEnsureState() {
  if (!state) return null;
  state.observation ||= {};
  const observation = state.observation;
  observation.version = OBSERVATION.VERSION;
  observation.quality ||= 'balanced';
  observation.events ||= [];
  observation.clock ||= 0;
  observation.followSelected ??= false;
  observation.showEvents ??= true;
  observation.mapScale ||= 1.34;
  observation.frameMs ||= 16.7;
  observation.lastGenerationCost ||= 0;
  return observation;
}

function observationFlushSave() {
  if (observationSaveHandle !== null) {
    if (typeof cancelIdleCallback === 'function') cancelIdleCallback(observationSaveHandle);
    else clearTimeout(observationSaveHandle);
  }
  observationSaveHandle = null;
  if (!observationSavePending) return;
  observationSavePending = false;
  observationLegacySaveState();
}

saveState = function observationDeferredSave(force = false) {
  if (!state) return;
  observationSavePending = true;
  if (force) { observationFlushSave(); return; }
  if (observationSaveHandle !== null) return;
  const callback = () => {
    observationSaveHandle = null;
    if (!observationSavePending) return;
    observationSavePending = false;
    observationLegacySaveState();
  };
  observationSaveHandle = typeof requestIdleCallback === 'function'
    ? requestIdleCallback(callback, { timeout: 900 })
    : setTimeout(callback, 240);
};

addEventListener('pagehide', observationFlushSave);
document.addEventListener('visibilitychange', () => { if (document.hidden) observationFlushSave(); });

function observationVisualBudget(meta) {
  const observation = observationEnsureState();
  const active = meta.demes.filter((deme) => deme.abundance > .25);
  const total = active.reduce((sum, deme) => sum + deme.abundance, 0);
  const modelCap = typeof MAX_ORGANISMS === 'number' ? MAX_ORGANISMS : 220;
  return Math.min(modelCap, OBSERVATION.qualityBudget(observation.quality, active.length, total));
}

metaDemeTargetCounts = function observationDemeTargetCounts() {
  const meta = metaEnsureState();
  if (!meta) return new Map();
  return OBSERVATION.allocateTargets(meta.demes, observationVisualBudget(meta));
};
