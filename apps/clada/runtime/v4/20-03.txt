            terminal.planetSpeciationMode = 'ring';
          }
        }
      }
    }
  }

  function seedCorridorColonization(community) {
    const planet = ensurePlanet(community);
    const generation = Number(community.generation) || 0;
    if (generation % 3 !== 0) return null;
    const patchMap = new Map((community.patches || []).map((patch) => [patch.id, patch]));
    const candidates = [];
    for (const deme of (community.demes || []).filter((entry) => entry.abundance > Math.max(7, (entry.carryingCapacity || 12) * .46))) {
      const source = patchMap.get(deme.patchId);
      if (!source) continue;
      for (const target of community.patches || []) {
        if (target.id === source.id) continue;
        const route = routeDescriptor(community, source, target, deme);
        if (!['current', 'land-bridge', 'ice-bridge'].includes(route.type) || route.modifier < .22) continue;
        const existing = (community.demes || []).find((entry) => entry.speciesId === deme.speciesId && entry.patchId === target.id && entry.guild === deme.guild && entry.abundance > .2);
        if (existing) continue;
        const score = route.modifier * (deme.traits?.migration || .4) * Math.min(1, deme.abundance / 18);
        candidates.push({ deme, source, target, route, score });
      }
    }
    candidates.sort((a, b) => b.score - a.score);
    const best = candidates[0];
    if (!best || randomAt(planet.seed ^ generation, 89) > Math.min(.16, best.score * .12)) return null;
    const moved = Math.min(best.deme.abundance * .08, 1.4 + best.score * 3.2);
    if (moved < .8) return null;
    best.deme.abundance -= moved;
    const newDeme = {
      id: community.nextDemeId++, speciesId: best.deme.speciesId, patchId: best.target.id, guild: best.deme.guild,
      abundance: moved, age: 0, traits: { ...best.deme.traits }, anchor: { ...best.deme.traits }, isolation: .02,
      geneFlow: .18 * best.route.modifier, lowYears: 0, founded: generation, colonizationSource: best.deme.id,
      planetRouteType: best.route.type, establishmentUntil: generation + 18
    };
    community.demes.push(newDeme);
    community.metrics.colonizations = (community.metrics.colonizations || 0) + 1;
    planet.metrics.corridorColonizations += 1;
    const event = { generation, type: 'planet-colonization', speciesId: newDeme.speciesId, patchId: newDeme.patchId, sourcePatchId: best.source.id, route: best.route.type, text: `Колонизация через ${best.route.type}` };
    community.events.push(event); planet.events.push(event);
    return newDeme;
  }

  function speciesDistance(community, a, b) {
    const vectorDistance = globalThis.CladaMetacommunityCore?.vectorDistance;
    return typeof vectorDistance === 'function' ? vectorDistance(a.traits || {}, b.traits || {}) : Math.sqrt(Object.keys(a.traits || {}).reduce((sum, key) => sum + ((a.traits[key] || 0) - (b.traits?.[key] || 0)) ** 2, 0) / Math.max(1, Object.keys(a.traits || {}).length));
  }

  function createHybridProposal(community, first, second) {
    if (!first || !second || first.speciesId === second.speciesId || first.patchId !== second.patchId) return null;
    const planet = ensurePlanet(community);
    const parentA = (community.species || []).find((species) => species.id === first.speciesId);
    const parentB = (community.species || []).find((species) => species.id === second.speciesId);
    if (!parentA || !parentB || parentA.extinct || parentB.extinct) return null;
    const founder = Math.min(3.4, first.abundance * .08, second.abundance * .08);
    if (founder < .9) return null;
    const traits = {};
    const keys = new Set([...Object.keys(first.traits || {}), ...Object.keys(second.traits || {})]);
    for (const key of keys) {
      const midpoint = ((first.traits?.[key] || 0) + (second.traits?.[key] || 0)) / 2;
      const transgressive = noiseAt(planet.seed, community.generation || 0, hashString(key)) * .025;
      traits[key] = clamp(midpoint + transgressive);
    }
    first.abundance -= founder * .55;
    second.abundance -= founder * .45;
    const deme = {
      id: community.nextDemeId++, speciesId: first.speciesId, patchId: first.patchId,
      guild: first.guild === second.guild ? first.guild : (traits.prey || 0) > .62 ? 'predator' : (traits.water || 0) > .62 ? 'filterer' : first.guild,
      abundance: founder, age: 0, traits, anchor: { ...traits }, isolation: .82, isolationAge: 40,
      geneFlow: 0, lowYears: 0, founded: community.generation, hybridParentId: second.speciesId, hybridOrigin: true
    };
    community.demes.push(deme);
    return { speciesId: first.speciesId, demeId: deme.id, mode: 'hybrid', secondParentId: second.speciesId, isolation: 1, geologicalCause: 'контактная гибридная зона' };
  }

  function createChromosomalProposal(community, source) {
    if (!source || source.abundance < 9 || source.guild === 'predator') return null;
    const founder = Math.min(2.8, source.abundance * .1);
    if (founder < 1) return null;
    source.abundance -= founder;
    const traits = { ...source.traits, fertility: clamp((source.traits?.fertility || .5) + .05), mateChoice: clamp((source.traits?.mateChoice || .4) + .16) };
    const deme = {
      id: community.nextDemeId++, speciesId: source.speciesId, patchId: source.patchId, guild: source.guild,
      abundance: founder, age: 0, traits, anchor: { ...traits }, isolation: .88, isolationAge: 44,
      geneFlow: 0, lowYears: 0, founded: community.generation, chromosomalOrigin: true
    };
    community.demes.push(deme);
    return { speciesId: source.speciesId, demeId: deme.id, mode: 'chromosomal', isolation: 1, geologicalCause: 'мгновенная репродуктивная изоляция' };
  }

  function extraProposals(community) {
    const planet = ensurePlanet(community);
    const generation = Number(community.generation) || 0;
    const mutation = clamp(community.env?.mutation || 0);
    if (generation - planet.lastHybridGeneration > 120) {
      const byPatch = new Map();
      for (const deme of (community.demes || []).filter((entry) => entry.abundance > 7)) {
        if (!byPatch.has(deme.patchId)) byPatch.set(deme.patchId, []);
        byPatch.get(deme.patchId).push(deme);
      }
      for (const demes of byPatch.values()) for (let i = 0; i < demes.length; i += 1) for (let j = i + 1; j < demes.length; j += 1) {
        const a = demes[i], b = demes[j];
        if (a.speciesId === b.speciesId) continue;
        const distance = speciesDistance(community, a, b);
        if (distance < .055 || distance > .24) continue;
        const chance = .00045 + mutation * .0018;
        if (randomAt(planet.seed ^ generation, a.id * 101 + b.id * 313) < chance) {
          const proposal = createHybridProposal(community, a, b);
          if (proposal) { planet.lastHybridGeneration = generation; planet.metrics.hybridOrigins += 1; return [proposal]; }
        }
      }
    }
    if (mutation > .58 && generation - planet.lastChromosomalGeneration > 150) {
      const candidates = (community.demes || []).filter((deme) => deme.abundance > 11 && deme.guild !== 'predator' && deme.guild !== 'scavenger');
      const source = candidates.sort((a, b) => b.abundance - a.abundance)[0];
      if (source && randomAt(planet.seed ^ generation, 109) < .0007 + mutation * .0013) {
        const proposal = createChromosomalProposal(community, source);
        if (proposal) { planet.lastChromosomalGeneration = generation; planet.metrics.chromosomalSplits += 1; return [proposal]; }
      }
    }
    return [];
  }

  function classifyProposal(community, proposal) {
    const deme = (community.demes || []).find((entry) => entry.id === proposal.demeId);
    const patch = (community.patches || []).find((entry) => entry.id === deme?.patchId);
    if (!deme || !patch) return proposal;
    if (proposal.secondParentId || deme.hybridOrigin) return { ...proposal, mode: 'hybrid' };
    if (deme.chromosomalOrigin) return { ...proposal, mode: 'chromosomal' };
    if (deme.ringTerminal || deme.planetSpeciationMode === 'ring') return { ...proposal, mode: 'ring', geologicalCause: 'расхождение вокруг географического барьера' };
    if (patch.island && (deme.age || 0) < 70 && (deme.abundance || 0) < 12) return { ...proposal, mode: 'peripatric', geologicalCause: 'эффект основателя на острове' };
    if ((deme.geographicBarrier || 0) > .74) return { ...proposal, mode: 'vicariant', geologicalCause: patch.mountainBarrier ? 'поднятие горного барьера' : 'разрыв ареала морем' };
    if ((deme.spatialIsolation || 0) < .3 && (deme.ecologicalDivergence || 0) > .09) return { ...proposal, mode: 'parapatric', geologicalCause: 'расхождение вдоль экологического градиента' };
    return proposal;
  }

  function decorateProposals(community, proposals = []) {
    return proposals.map((proposal) => classifyProposal(community, proposal));
  }

  function compact(community) {
    const planet = ensurePlanet(community);
    return {
      schema: 'clada-living-planet-v1', epoch: epochFor(planet).id, epochLabel: epochFor(planet).label,
      seaLevel: round(planet.seaLevel), climate: round(planet.climate), humidity: round(planet.humidity),
      glaciation: round(planet.glaciation), tectonics: round(planet.tectonics),
      metrics: { ...planet.metrics }, timeline: planet.timeline.slice(-180), events: planet.events.slice(-120),
      patches: (community.patches || []).map((patch) => ({ id: patch.id, water: patch.water ? 1 : 0, coast: patch.coast ? 1 : 0, island: patch.island ? 1 : 0, biome: patch.biome, elevation: round(patch.elevation), temperature: round(patch.temperature), moisture: round(patch.moisture), seaIce: patch.seaIce ? 1 : 0, mountain: patch.mountainBarrier ? 1 : 0 })),
      routes: Object.fromEntries(Object.entries(planet.routes).filter(([, route]) => route.open).slice(0, 48))
    };
  }

  return {
    VERSION, EPOCHS, clamp, lerp, round, hash, hashString, randomAt, noiseAt, epochFor,
    ensurePlanet, biomeFor, dynamicPatch, transformPoint, prepareGeneration, routeDescriptor,
    connectivityModifier, areLinked, applyHabitatStress, adjustIsolation, seedCorridorColonization,
    createHybridProposal, createChromosomalProposal, extraProposals, classifyProposal, decorateProposals, compact
  };
