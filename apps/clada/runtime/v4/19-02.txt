function fieldDownload(payload, fileName, message) {
  const source = typeof payload === 'string' ? payload : JSON.stringify(payload, null, 2);
  const blob = new Blob([source], { type: 'application/json' });
  const url = URL.createObjectURL(blob); const anchor = document.createElement('a');
  anchor.href = url; anchor.download = fileName; anchor.click();
  setTimeout(() => URL.revokeObjectURL(url), 1200);
  showToast(`${message} · ${Math.ceil(source.length / 1024)} КБ`);
}

function fieldBuildReplayBundle() {
  return FIELD.buildReplayBundle({
    seed: state.seed,
    originMode: state.originMode || LIVING.originMode,
    commands: state.commandLog || [],
    version: FIELD_APP_VERSION,
    createdGeneration: state.generation
  });
}

buildCompactDiagnostic = function fieldBuildCompactDiagnostic() {
  const payload = fieldLegacyBuildCompactDiagnostic();
  payload.version = FIELD_APP_VERSION;
  payload.journal = {
    schema: 'clada-field-journal-v1',
    alerts: fieldEnsureState().alerts.slice(-80),
    speciesSeries: Object.fromEntries(Object.entries(fieldEnsureState().speciesSeries).map(([id, series]) => [id, FIELD.compactSeries(series, 120)])),
    replay: fieldBuildReplayBundle()
  };
  return payload;
};

function fieldBuildSpeciesDiagnostic(speciesId) {
  const base = buildCompactDiagnostic();
  const related = new Set([Number(speciesId)]);
  for (const species of base.species) if (species.parent === Number(speciesId) || species.id === Number(speciesId)) related.add(species.id);
  const focal = state.species.find((entry) => entry.id === Number(speciesId));
  if (focal?.parentId) related.add(focal.parentId);
  return {
    schema: 'clada-species-diagnostic-v1',
    app: base.app,
    version: FIELD_APP_VERSION,
    world: base.world,
    summary: base.summary,
    focalSpeciesId: Number(speciesId),
    explanation: (() => { const value = FIELD.pressureBreakdown(state.metacommunity, speciesId); return { direction: value.direction, headline: value.headline, factors: value.factors }; })(),
    species: base.species.filter((entry) => related.has(entry.id)),
    demes: base.demes.filter((entry) => related.has(entry.species)),
    patches: base.patches,
    series: FIELD.compactSeries(fieldSpeciesSeries(speciesId), 180),
    replay: fieldBuildReplayBundle(),
    events: base.events.filter((event) => related.has(Number(event.speciesId))).slice(-120),
    mortality: Object.fromEntries(Object.entries(base.mortality || {}).filter(([id]) => related.has(Number(id))))
  };
}

}

function fieldExportReplay() {
  fieldDownload(fieldBuildReplayBundle(), `clada-replay-g${state.generation}.creplay.json`, 'Сценарий эксперимента экспортирован');
}

function fieldApplyReplayCommand(command) {
  if (!command) return;
  fieldApplyingReplay = true;
  try {
    if (command.type === 'environment' && ['temperature', 'food', 'mutation'].includes(command.key)) {
      state.env[command.key] = clamp(Number(command.value));
      syncPressureUI();
    } else if (command.type === 'cataclysm' && command.value) {
      applyCataclysm(command.value);
    } else if (command.type === 'introduction') {
      seedAt(Number(command.x) || .5, Number(command.y) || .5, Math.max(1, Number(command.amount) || 10));
    }
  } finally {
    fieldApplyingReplay = false;
  }
  state.commandLog ||= [];
  state.commandLog.push({ ...command });
  state.commandLog = state.commandLog.slice(-240);
}

function fieldProcessReplay() {
  const replay = fieldEnsureState()?.replay;
  if (!replay?.active) return;
  while (replay.cursor < replay.commands.length && replay.commands[replay.cursor].g <= state.generation) {
    fieldApplyReplayCommand(replay.commands[replay.cursor]);
    replay.cursor += 1;
  }
  if (replay.cursor >= replay.commands.length) {
    replay.active = false;
    showToast('Воспроизведение завершено');
  }
}

function fieldStartReplay(bundle) {
  if (!bundle || bundle.schema !== 'clada-replay-v1' || !Array.isArray(bundle.commands)) throw new Error('Неверный формат сценария');
  LIVING.originMode = bundle.originMode === 'primordial' ? 'primordial' : 'mature';
  fieldLegacyBuildWorld(bundle.seed || 'garden');
  const journal = fieldEnsureState();
  journal.speciesSeries = {};
  journal.alerts = [];
  journal.replay = { active: true, cursor: 0, commands: bundle.commands.map((entry) => ({ ...entry })).sort((a, b) => a.g - b.g), sourceVersion: bundle.version || null };
  state.commandLog = [{ g: state.generation, type: 'world-created', seed: bundle.seed || 'garden', originMode: LIVING.originMode }];
  fieldProcessReplay();
  fieldRecordSnapshot(true);
  state.paused = false;
  syncAllUI();
  saveState();
  showToast('Эксперимент запущен заново');
}

function fieldInstallInterface() {
  fieldEnsureState(); fieldRecordSnapshot(true);
  if (!viewTabs.querySelector('[data-field-journal]')) viewTabs.insertAdjacentHTML('beforeend', '<button type="button" data-field-journal>ЖУРНАЛ</button>');
  if (!document.querySelector('#fieldReplayInput')) document.querySelector('.app').insertAdjacentHTML('beforeend', '<input id="fieldReplayInput" type="file" accept="application/json,.json,.creplay.json" hidden>');
  viewTabs.addEventListener('click', (event) => { const button = event.target.closest('[data-field-journal]'); if (!button) return; event.preventDefault(); fieldOpenJournal('overview'); tone(390, .05, .015, 'triangle'); });
  const replayInput = document.querySelector('#fieldReplayInput');
  replayInput.addEventListener('change', () => {
    const file = replayInput.files?.[0]; replayInput.value = ''; if (!file) return;
    const reader = new FileReader(); reader.onload = () => { try { fieldStartReplay(JSON.parse(String(reader.result))); } catch (error) { showToast(error.message || 'Не удалось открыть сценарий', 3000); } }; reader.readAsText(file);
  });
}

sheetBody.addEventListener('click', (event) => {
  const speciesButton = event.target.closest('[data-field-species]');
  if (speciesButton) { event.preventDefault(); event.stopImmediatePropagation(); openSpeciesSheet(state.species.find((entry) => entry.id === Number(speciesButton.dataset.fieldSpecies))); return; }
  const panel = event.target.closest('[data-field-panel]');
  if (panel) { event.preventDefault(); event.stopImmediatePropagation(); fieldOpenJournal(panel.dataset.fieldPanel); return; }
  const action = event.target.closest('[data-field-action]');
  if (!action) return;
  event.preventDefault(); event.stopImmediatePropagation();
  const name = action.dataset.fieldAction;
  if (name === 'journal') fieldOpenJournal('overview');
  else if (name === 'export-replay') fieldExportReplay();
  else if (name === 'import-replay') document.querySelector('#fieldReplayInput').click();
  else if (name === 'repeat') fieldConfirmRepeat();
  else if (name === 'repeat-now') { try { fieldStartReplay(fieldPendingReplayBundle); fieldPendingReplayBundle = null; } catch (error) { showToast(error.message || 'Не удалось повторить эксперимент'); } }
  else if (name === 'cancel') closeSheet();
  else if (name === 'species-export') fieldExportSpecies(Number(action.dataset.speciesId));
}, true);

fieldInstallInterface();
