      const median = sorted[Math.floor(sorted.length / 2)];
      const p90 = sorted[Math.floor(sorted.length * .9)];
      observation.frameMs = Math.round(median * 10) / 10;
      const previous = observation.quality;
      if (p90 > 31 || median > 24) observation.quality = 'low';
      else if (p90 < 21 && median < 18) observation.quality = 'high';
      else observation.quality = 'balanced';
      observationFrameSamples = [];
      if (previous !== observation.quality) metaReconcileRepresentatives({ soft: true });
    }
    observationFollowSelected();
    if ((state.step || 0) % 10 === 0) observationUpdateHud();
  }
  requestAnimationFrame(observationMonitorFrame);
}
