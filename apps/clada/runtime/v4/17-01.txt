function metaRound(value, digits = 3) {
  const factor = 10 ** digits;
  return Math.round((Number(value) || 0) * factor) / factor;
}

function metaPatchIdAt(x, y) {
  return `${Math.min(3, Math.max(0, Math.floor(x * 4)))}${Math.min(2, Math.max(0, Math.floor(y * 3)))}`;
}

function metaBuildPatches() {
  const patches = [];
  for (let by = 0; by < 3; by += 1) for (let bx = 0; bx < 4; bx += 1) {
    const x = (bx + .5) / 4;
    const y = (by + .5) / 3;
    const terrain = terrainAt(x, y);
    patches.push({
      id: `${bx}${by}`, bx, by, x, y,
      biome: terrain.biome, water: Boolean(terrain.water), coast: Boolean(terrain.coast),
      temperature: terrain.temperature, moisture: terrain.moisture, elevation: terrain.elevation,
      region: terrain.region, neighbors: []
    });
  }
  for (const patch of patches) {
    patch.neighbors = patches
      .filter((candidate) => candidate !== patch && Math.abs(candidate.bx - patch.bx) <= 1 && Math.abs(candidate.by - patch.by) <= 1)
      .map((candidate) => candidate.id);
  }
  return patches;
}

function metaPatchById(id) {
  return state.metacommunity?.patches?.find((patch) => patch.id === id) || null;
}

function metaPointInPatch(patch, preferWater = patch?.water) {
  if (!patch) return { x: .5, y: .5 };
  let best = { x: patch.x, y: patch.y, score: -Infinity };
  for (let attempt = 0; attempt < 36; attempt += 1) {
    const x = clamp((patch.bx + randomRange(.08, .92)) / 4, .025, .975);
    const y = clamp((patch.by + randomRange(.08, .92)) / 3, .04, .96);
    const terrain = terrainAt(x, y);
    const waterScore = terrain.water === preferWater ? 1 : -.5;
    const score = waterScore + (terrain.biome === patch.biome ? .5 : 0) + random() * .1;
    if (score > best.score) best = { x, y, score };
  }
  return best;
}

function metaTraitsFromGenome(genome) {
  enrichGenome(genome);
  return META.traitVector(genome);
}

function metaGenomeFromTraits(traits, baseGenome = null) {
  const base = deepClone(baseGenome || makeGenome());
  enrichGenome(base);
  base.thermal = clamp(traits.thermal);
  base.moisture = clamp(traits.moisture);
  base.altitude = clamp(traits.altitude);
  base.waterAffinity = clamp(traits.water);
  base.aquatic = clamp(traits.water * .92);
  base.size = clamp(.42 + clamp(traits.size) * 1.3, .42, 1.72);
  base.speed = clamp(.28 + clamp(traits.speed) * 1.52, .28, 1.8);
  base.preyDiet = clamp(traits.prey);
  base.plantDiet = clamp(traits.plant);
  base.filterDiet = clamp(traits.filter);
  base.carrionDiet = clamp(traits.carrion);
  base.diet = clamp(traits.prey * .78 + traits.carrion * .32 + (1 - traits.plant) * .12);
  base.nocturnal = clamp(traits.nocturnal);
  base.mateChoice = clamp(traits.mateChoice);
  base.migration = clamp(traits.migration);
  base.fertility = clamp(traits.fertility * 1.7, .35, 1.7);
  base.parentalCare = clamp(traits.parentalCare);
  base.armor = clamp(traits.armor);
  base.cooperation = clamp(traits.social);
  base.social = clamp(traits.social);
  if (traits.water > .62) base.fins = Math.max(base.fins || 0, .66);
  if (traits.water < .24) base.fins = Math.min(base.fins || 0, .3);
  if (typeof stabilityInferRaw === 'function') stabilityInferRaw(base);
  if (typeof stabilityExpressGenome === 'function') stabilityExpressGenome(base);
  return enrichGenome(base);
}

function metaSpeciesFor(appSpecies) {
  return state.metacommunity?.species?.find((species) => species.id === appSpecies?.id) || null;
}

function metaDemeForOrganism(organism) {
  const meta = state.metacommunity;
  if (!meta) return null;
  if (organism.macroDemeId) {
    const direct = meta.demes.find((deme) => deme.id === organism.macroDemeId && deme.abundance > .02);
    if (direct) return direct;
  }
  const patchId = metaPatchIdAt(organism.x, organism.y);
  const appSpecies = state.species.find((species) => species.id === organism.speciesId);
  const guild = META.guildFromGenome(organism.genome, appSpecies?.niche);
  return meta.demes.find((deme) => deme.speciesId === organism.speciesId && deme.patchId === patchId && deme.guild === guild && deme.abundance > .02)
    || meta.demes.filter((deme) => deme.speciesId === organism.speciesId && deme.abundance > .02).sort((a, b) => b.abundance - a.abundance)[0]
    || null;
}

function metaBuildCommunityFromState() {
  const patches = metaBuildPatches();
  const patchMap = new Map(patches.map((patch) => [patch.id, patch]));
  const livingIds = new Set(state.organisms.map((organism) => organism.speciesId));
  const species = state.species
    .filter((entry) => livingIds.has(entry.id))
    .map((entry) => {
      const traits = metaTraitsFromGenome(entry.centroid);
      const guild = META.guildFromGenome(entry.centroid, entry.niche);
      return {
        id: entry.id, parentId: entry.parentId, born: entry.born, extinct: false,
        guild, traits: deepClone(traits), anchor: deepClone(traits), niche: entry.niche || guild,
        extinctionDebt: 0, lastSpeciationGeneration: entry.lastSpeciationGeneration ?? entry.born ?? 0,
        appSpeciesId: entry.id
      };
    });
  const groups = new Map();
  for (const organism of state.organisms) {
    const appSpecies = state.species.find((entry) => entry.id === organism.speciesId);
    if (!appSpecies) continue;
    const patchId = metaPatchIdAt(organism.x, organism.y);
    const guild = META.guildFromGenome(organism.genome, appSpecies.niche);
    const key = `${organism.speciesId}:${patchId}:${guild}`;
    if (!groups.has(key)) groups.set(key, { speciesId: organism.speciesId, patchId, guild, members: [] });
    groups.get(key).members.push(organism);
  }
  const demes = [];
  let nextDemeId = 1;
  for (const group of groups.values()) {
    const speciesEntry = species.find((entry) => entry.id === group.speciesId);
    if (!speciesEntry) continue;
    const traits = group.members.length ? metaTraitsFromGenome(livingCentroid(group.members)) : deepClone(speciesEntry.traits);
    const oldPopulation = state.populations?.find((population) => population.speciesId === group.speciesId && population.size > 0 && metaPatchIdAt(population.center?.x ?? group.members[0]?.x ?? .5, population.center?.y ?? group.members[0]?.y ?? .5) === group.patchId);
    const scale = state.metacommunity?.sampleScale || 1.55;
    const abundance = Math.max(2.8, oldPopulation?.macroAbundance || group.members.length * scale);
    const deme = {
      id: nextDemeId++, speciesId: group.speciesId, patchId: group.patchId, guild: group.guild,
      abundance, age: Math.max(4, state.generation - (oldPopulation?.founded ?? state.generation - 8)),
      traits: deepClone(traits), anchor: deepClone(traits), isolation: (oldPopulation?.speciationProgress || 0) * .55,
      isolationAge: oldPopulation?.isolation || 0, geneFlow: oldPopulation?.geneFlowEMA || 0,
      lowYears: 0, founded: oldPopulation?.founded ?? state.generation, appPopulationId: oldPopulation?.id || null
    };
    demes.push(deme);
    for (const organism of group.members) organism.macroDemeId = deme.id;
  }
  const community = {
    modelVersion: META_MODEL_VERSION, generation: state.generation, rng: state.rng || 123456789,
    env: deepClone(state.env), patches, species, demes, resources: {}, events: [], nextDemeId,
    metrics: { speciations: 0, localExtinctions: 0, colonizations: 0 },
    nextSpeciesId: state.nextSpeciesId, originSeed: state.seed, originMode: state.originMode || LIVING.originMode,
    patchLookup: Object.fromEntries([...patchMap].map(([id, patch]) => [id, patch.region]))
  };
  META.ensureCommunity(community);
  return community;
}

function metaEnsureState() {
  if (!state) return null;
  state.commandLog ||= [];
  state.macroTimeline ||= [];
  if (!state.metacommunity || state.metacommunity.modelVersion !== META_MODEL_VERSION) {
    state.metacommunity = metaBuildCommunityFromState();
  }
  state.metacommunity.env = deepClone(state.env);
  return META.ensureCommunity(state.metacommunity);
}
