function livingHash(seed, x, y, salt = 0) {
  const value = Math.sin((x * 127.1 + y * 311.7 + seed * .013 + salt * 74.7)) * 43758.5453;
  return value - Math.floor(value);
}

function terrainAt(x, y) {
  const seed = state?.terrainSeed || 1729;
  const basin = Math.sin((x + seed * .0003) * Math.PI * 2.2) * .13;
  const ridge = Math.cos((y - seed * .00017) * Math.PI * 3.1) * .09;
  const continental = .5 - Math.hypot(x - .52, y - .52) * .34;
  const detail = (livingHash(seed, x * 4, y * 4, 1) - .5) * .16;
  let elevation = clamp(continental + basin + ridge + detail, 0, 1);
  if (state?.seed === 'rift') elevation = clamp(elevation + Math.abs(x - .5) * .13 - .03, 0, 1);
  if (state?.seed === 'red') elevation = clamp(elevation - Math.max(0, .18 - Math.hypot(x - .72, y - .42)) * 1.4, 0, 1);
  const waterLine = state?.seed === 'garden' ? .39 : state?.seed === 'rift' ? .43 : .42;
  const wetSeason = .5 + Math.sin(((state?.season || 0) / 12) * TAU) * .5;
  const riverCenter = .46 + Math.sin(y * 9 + seed * .001) * .075;
  const river = Math.abs(x - riverCenter) < .008 + wetSeason * .012 && elevation < .67;
  const water = elevation < waterLine || river;
  const coast = !water && (elevation < waterLine + .07 || Math.abs(x - riverCenter) < .035);
  const moistureNoise = livingHash(seed, x * 3.3, y * 3.3, 2);
  const moisture = clamp((1 - Math.abs(x - .48)) * .25 + moistureNoise * .55 + (water || coast ? .32 : 0) - elevation * .18);
  const temperature = clamp((state?.env?.temperature ?? .52) + (y - .5) * .27 - Math.max(0, elevation - .67) * .48);
  let biome = 'grassland';
  if (water) biome = elevation < waterLine - .1 ? 'ocean' : 'shallows';
  else if (temperature < .16) biome = 'snow';
  else if (elevation > .73) biome = 'highland';
  else if (moisture > .72) biome = coast ? 'wetland' : 'forest';
  else if (moisture < .22 && temperature > .58) biome = 'desert';
  else if (moisture < .38) biome = 'scrub';
  const bx = Math.min(3, Math.floor(x * 4));
  const by = Math.min(2, Math.floor(y * 3));
  const hydro = water ? (x < .5 ? 'W' : 'E') : coast ? 'C' : 'L';
  const barrier = elevation > .78 ? 'M' : hydro;
  return { elevation, moisture, temperature, water, coast, biome, region: `${bx}${by}${barrier}` };
}

function enrichGenome(genome) {
  if (!genome) return genome;
  gv(genome);
  const derived = {
    plantDiet: clamp(1 - genome.diet * .83),
    preyDiet: clamp((genome.diet - .25) * 1.34),
    carrionDiet: clamp(.12 + genome.diet * .52 + genome.armor * .08),
    filterDiet: clamp((genome.aquatic || 0) * .68 + (1 - genome.size) * .16),
    brain: clamp((genome.vision - .35) * .42 + (genome.social || 0) * .38),
    stamina: clamp(.35 + genome.speed * .24 - genome.size * .08),
    parentalCare: clamp((genome.social || 0) * .48 + (1 - genome.fertility / 1.7) * .35),
    brood: clamp(genome.fertility / 1.7 * .82 + .08),
    reproStrategy: clamp(.35 + (genome.social || 0) * .35 + genome.pattern * .18),
    mateChoice: clamp((genome.display || 0) * .52 + (genome.vision / 1.75) * .32),
    territory: clamp(genome.diet * .42 + genome.armor * .18 - (genome.social || 0) * .2),
    ambush: clamp(genome.diet * .38 + (1 - genome.speed / 1.8) * .23 + (1 - genome.vision / 1.75) * .08),
    migration: clamp(genome.speed * .25 + (genome.stamina || .5) * .34 + (genome.aquatic || 0) * .12),
    nocturnal: clamp((360 - genome.hue) / 720 + (1 - genome.thermal) * .18),
    cooperation: clamp((genome.social || 0) * .74 + genome.brain * .18),
    toxinResistance: clamp(genome.armor * .28 + genome.metabolism * .15),
    altitude: clamp((1 - genome.aquatic) * .35 + (1 - genome.thermal) * .25),
    moisture: clamp((genome.aquatic || 0) * .65 + (genome.fur || 0) * .1 + .18),
    waterAffinity: clamp((genome.aquatic || 0) * .82 + (genome.fins || 0) * .2),
    burrow: clamp((1 - genome.size / 1.72) * .38 + genome.armor * .17)
  };
  for (const key of LIVING.genomeKeys) if (!Number.isFinite(genome[key])) genome[key] = derived[key];
  return genome;
}

function applyTradeoffs(genome) {
  enrichGenome(genome);
  const flight = genome.wing || 0;
  const shell = genome.shell || 0;
  genome.armor = clamp(genome.armor - flight * .12 + shell * .08);
  genome.speed = clamp(genome.speed - shell * .17 - genome.fur * .04, .28, 1.8);
  genome.metabolism = clamp(genome.metabolism + flight * .08 + genome.brain * .11 + genome.display * .045, .38, 1.65);
  genome.fertility = clamp(genome.fertility - genome.parentalCare * .26 - genome.size * .055, .35, 1.7);
  genome.brood = clamp(genome.brood - genome.parentalCare * .42 + (1 - genome.size / 1.72) * .08);
  genome.cooperation = clamp(genome.cooperation - genome.territory * .24);
  return genome;
}

makeGenome = function livingMakeGenome(base = null, mutationScale = 0) {
  const source = base ? enrichGenome({ ...base }) : null;
  const genome = livingLegacyMakeGenome(source, mutationScale);
  enrichGenome(genome);
  if (source) {
    const magnitude = .012 + mutationScale * .14;
    for (const key of LIVING.genomeKeys) genome[key] = clamp(source[key] + gauss() * magnitude);
  }
  return applyTradeoffs(genome);
};
