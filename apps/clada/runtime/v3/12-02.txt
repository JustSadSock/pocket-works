    }

  }

    ctx.fillStyle = 'rgba(31,42,36,.78)'; ctx.font = '700 11px ui-sans-serif,system-ui'; ctx.textAlign = 'center';

  }
  ctx.restore(); hits(camera);
  ctx.fillStyle = 'rgba(231,225,207,.84)'; ctx.fillRect(cssWidth - 112, cssHeight - 28, 101, 18);
  ctx.fillStyle = '#1f2a24'; ctx.font = '700 8px ui-sans-serif,system-ui'; ctx.textAlign = 'right';

};

function livingTreeLayout(generation) {
  const visible = state.species.filter((species) => species.born <= generation).slice(-72);
  const yMap = new Map(), children = new Map();
  for (const species of visible) {
    const parent = visible.some((candidate) => candidate.id === species.parentId) ? species.parentId : null;
    if (!children.has(parent)) children.set(parent, []);
    children.get(parent).push(species);
  }
  let leaf = 0;
  function assign(species) {
    const descendants = children.get(species.id) || [];
    if (!descendants.length) { yMap.set(species.id, leaf); leaf += 1; return leaf - 1; }
    const ys = descendants.map(assign); const value = ys.reduce((sum, y) => sum + y, 0) / ys.length;
    yMap.set(species.id, value); return value;
  }
  (children.get(null) || []).forEach(assign);
  for (const species of visible) if (!yMap.has(species.id)) { yMap.set(species.id, leaf); leaf += 1; }
  return { visible, yMap, rows: Math.max(1, leaf) };
}
