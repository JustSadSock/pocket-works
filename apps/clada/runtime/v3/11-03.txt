const region = terrainAt(midpoint.x, midpoint.y).region;
  const population = livingPopulationAt(parent.speciesId, region, baseGenome);
  if (hybrid) { population.hybrid = true; population.hybridParentId = mate.speciesId; }
  for (let index = 0; index < available; index += 1) {
    const genome = index === 0 ? baseGenome : makeGenome(baseGenome, mutationScale * .35);
    const angle = randomRange(0, TAU);
    const careBoost = genome.parentalCare * 2.6;
    const child = makeOrganism({
      x: midpoint.x + Math.cos(angle) * randomRange(.004, .018),
      y: midpoint.y + Math.sin(angle) * randomRange(.004, .018),
      genome, speciesId: parent.speciesId,
      generation: Math.max(parent.generation, mate?.generation || parent.generation) + 1,
      energy: 5.8 + careBoost, parentId: parent.id
    });
    child.secondParentId = mate?.id || null;
    child.hybrid = hybrid;
    child.populationId = population.id;
    child.homeRegion = region;
    child.sex = genome.reproStrategy < .18 ? 'none' : random() < .5 ? 'f' : 'm';
    child.lifeStage = 'juvenile';
    state.organisms.push(child);
    state.statistics.births += 1;
    if (hybrid) state.statistics.hybrids += 1;
  }
  const reproductiveCost = .74 + baseGenome.parentalCare * .07 + available * .025;
  parent.energy *= reproductiveCost;
  parent.age += 22 + baseGenome.parentalCare * 26;
  if (mate) { mate.energy *= .8 + baseGenome.parentalCare * .04; mate.age += 14; }
  if (parentPopulation) parentPopulation.lastSeen = state.generation;
};

function livingFeed(organism, resources, resourceIndex) {
  const genome = organism.genome;
  const bucket = interactionBucket(organism.speciesId);
  if (resources.plant && resources.plantDistance < (.008 + genome.size * .004) ** 2 && genome.plantDiet > .16) {
    const bite = Math.min(resources.plant.energy, (.1 + genome.size * .16) * genome.plantDiet);
    resources.plant.energy -= bite;
    organism.energy += bite * (1.08 + genome.plantDiet * .42);
    organism.lastMeal = state.step;
    bucket.plants += bite;
  }
  if (resources.carrion && resources.carrionDistance < (.009 + genome.size * .004) ** 2 && genome.carrionDiet > .16) {
    const bite = Math.min(resources.carrion.energy, (.12 + genome.size * .13) * genome.carrionDiet);
    resources.carrion.energy -= bite;
    organism.energy += bite * (1.05 + genome.toxinResistance * .32);
    organism.lastMeal = state.step;
    bucket.carrion += bite;
  }
  if (resources.prey && resources.preyDistance < (.007 + genome.size * .005) ** 2 && genome.preyDiet > .2) {
    const packBonus = 1 + genome.cooperation * .22;
    const damage = (.045 + genome.size * .085) * genome.preyDiet * packBonus * (1 + genome.ambush * .18) * (1.08 - resources.prey.genome.armor * .52);
    resources.prey.energy -= damage;
    organism.energy += damage * .7;
    organism.lastMeal = state.step;
    livingIncrement(bucket.prey, resources.prey.speciesId, damage);
    livingIncrement(interactionBucket(resources.prey.speciesId).predators, organism.speciesId, damage);
    if (resources.prey.energy <= 0) resources.prey.cause = 'хищничество';
  }
  const terrain = terrainAt(organism.x, organism.y);
  if (terrain.water && genome.filterDiet > .22) {
    const micro = resourceIndex.micro.nearby(organism.x, organism.y, .05).find((entry) => entry.energy > .1 && distanceSq(organism.x, organism.y, entry.x, entry.y) < .045 ** 2);
    if (micro) {
      const bite = Math.min(micro.energy, .015 + genome.filterDiet * .035);
      micro.energy -= bite;
      organism.energy += bite * 1.8;
      bucket.filter += bite;
    }
  }
}

function livingMovementAndCost(organism) {
  const genome = organism.genome;
  const terrain = terrainAt(organism.x, organism.y);
  const fitness = habitatFitness(genome, terrain);
  const dayPhase = (state.step % GENERATION_STEPS) / GENERATION_STEPS;
  const activePhase = genome.nocturnal > .55 ? (dayPhase > .55 || dayPhase < .12) : (dayPhase > .08 && dayPhase < .72);
  const activity = activePhase ? 1 : .48;
  const wander = .000009 * (1.5 - genome.vision * .27) * activity;
  organism.vx += Math.cos(organism.id * 2.13 + state.step * .021 + organism.y * 9) * wander;
  organism.vy += Math.sin(organism.id * 1.71 + state.step * .018 + organism.x * 11) * wander;
  const terrainSpeed = terrain.water ? lerp(.42, 1.15, genome.waterAffinity) : lerp(1.08, .48, genome.waterAffinity);
  const maxVelocity = (.00034 + genome.speed * .0005 + genome.stamina * .00008) * terrainSpeed * activity * (1 - genome.ambush * .12);
  const velocity = Math.hypot(organism.vx, organism.vy) || 1;
  if (velocity > maxVelocity) { organism.vx = organism.vx / velocity * maxVelocity; organism.vy = organism.vy / velocity * maxVelocity; }
  organism.vx *= .984; organism.vy *= .984;
  organism.x += organism.vx; organism.y += organism.vy;
  if (organism.x < .018 || organism.x > .982) { organism.x = clamp(organism.x, .018, .982); organism.vx *= -1.35; }
  if (organism.y < .03 || organism.y > .97) { organism.y = clamp(organism.y, .03, .97); organism.vy *= -1.35; }

  const currentTerrain = terrainAt(organism.x, organism.y);
  const climateMismatch = Math.abs(livingLocalTemperature(organism.x, organism.y) - genome.thermal);
  const movementCost = (.00145 + genome.speed ** 2 * .00105 + genome.brain * .0005 + genome.wing * .00045 + genome.shell * .00042) * genome.metabolism * activity;
  const habitatCost = (1 - habitatFitness(genome, currentTerrain)) ** 1.3 * .0065;
  const climateCost = climateMismatch ** 1.45 * .006 * (1.16 - genome.fur * .24);
  organism.energy -= movementCost + habitatCost + climateCost + applyShockCost(organism);
  if (fitness < .2 && organism.energy <= 0) organism.cause = currentTerrain.water ? 'неподходящая водная среда' : 'неподходящая суша';
}
