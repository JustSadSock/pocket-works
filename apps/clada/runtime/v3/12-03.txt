drawStrata = function livingDrawStrata() {
  ctx.fillStyle = '#ddd5bf'; ctx.fillRect(0, 0, cssWidth, cssHeight);
  const history = state.history; renderHits = [];
  if (!history.length) return;
  const layerHeight = Math.max(7, (cssHeight - 48) / history.length);
  const populationMax = Math.max(1, ...history.map((entry) => entry.population));
  for (let index = 0; index < history.length; index += 1) {
    const snapshot = history[index], y = cssHeight - 24 - (index + 1) * layerHeight;
    const richness = Object.values(snapshot.counts || {}).filter((count) => count > 0).length;
    const season = snapshot.season ?? snapshot.generation % 12;
    const hue = lerp(205, 27, snapshot.env.temperature);
    ctx.fillStyle = `hsl(${hue} 20% ${78 - snapshot.population / populationMax * 12}%)`;
    ctx.fillRect(0, y, cssWidth, layerHeight + .6);
    ctx.fillStyle = 'rgba(54,81,59,.43)';
    ctx.fillRect(54, y + layerHeight * .24, snapshot.population / populationMax * (cssWidth - 104), Math.max(1, layerHeight * .5));
    for (let dot = 0; dot < Math.min(18, richness); dot += 1) {
      ctx.fillStyle = `rgba(154,89,70,${.32 + dot / 22})`;
      ctx.beginPath(); ctx.arc(cssWidth - 16 - dot * 4.4, y + layerHeight / 2, 1.5, 0, TAU); ctx.fill();
    }
    if (snapshot.event) {
      ctx.fillStyle = snapshot.event.type === 'extinction' || snapshot.event.type === 'shock' ? '#9a5946' : '#36513b';
      ctx.beginPath(); ctx.moveTo(44, y + layerHeight / 2); ctx.lineTo(50, y + layerHeight / 2 - 4); ctx.lineTo(50, y + layerHeight / 2 + 4); ctx.closePath(); ctx.fill();
    }
    if (index % Math.max(1, Math.ceil(history.length / 8)) === 0 || index === history.length - 1) {
      ctx.fillStyle = 'rgba(31,42,36,.72)'; ctx.font = '8px ui-sans-serif,system-ui'; ctx.textAlign = 'left'; ctx.fillText(`G${snapshot.generation}`, 7, y + layerHeight * .72);
      ctx.textAlign = 'right'; ctx.fillText(SEASONS[season]?.split(' ').at(-1) || '', cssWidth - 7, y + layerHeight * .72);
    }
    renderHits.push({ type: 'history', id: index, x: cssWidth / 2, y: y + layerHeight / 2, radius: Math.max(8, layerHeight / 2) });
  }
  const selectedIndex = state.fossilIndex ?? history.length - 1;
  const selectedY = cssHeight - 24 - (selectedIndex + .5) * layerHeight;
  ctx.strokeStyle = '#1f2a24'; ctx.lineWidth = 1.5; ctx.beginPath(); ctx.moveTo(0, selectedY); ctx.lineTo(cssWidth, selectedY); ctx.stroke();
  ctx.fillStyle = '#1f2a24'; ctx.font = '600 12px ui-serif,Georgia,serif'; ctx.textAlign = 'left'; ctx.fillText('ЭКОЛОГИЧЕСКАЯ ЛЕТОПИСЬ', 8, 16);
};

function livingSpeciesStatus(species, count) {
  if (species.extinct !== null) return { label: 'ВЫМЕР', tone: 'extinct' };
  if (count <= 3) return { label: 'НА ГРАНИ', tone: 'danger' };
  if (species.trend < -3) return { label: 'СОКРАЩАЕТСЯ', tone: 'warning' };
  if (species.trend > 3) return { label: 'РАСШИРЯЕТСЯ', tone: 'growing' };
  return { label: 'УСТОЙЧИВ', tone: 'stable' };
}

function livingSpeciesById(id) { return state.species.find((species) => species.id === id) || null; }
function livingNameById(id) { return livingSpeciesById(id)?.common || 'не зафиксирован'; }


function livingMainMortality(speciesId) {
  const entries = Object.entries(state.mortality?.[speciesId] || {}).sort((a, b) => b[1] - a[1]);
  return entries[0]?.[0] || 'данных пока мало';
}

function livingClosestRelative(species) {
  return state.species
    .filter((candidate) => candidate.id !== species.id && candidate.extinct === null)
    .map((candidate) => ({ species: candidate, distance: genomeDistance(species.centroid, candidate.centroid) }))
    .sort((a, b) => a.distance - b.distance)[0] || null;
}

function livingPopulationRows(speciesId) {
  const populations = state.populations.filter((population) => population.speciesId === speciesId && population.size > 0).sort((a, b) => b.size - a.size);
  if (!populations.length) return '<div class="empty-mini">Локальные популяции не сохранились.</div>';
  return populations.map((population) => {
    const terrain = terrainAt(Number(population.region[0]) / 4 + .12, Number(population.region[1]) / 3 + .16);
    const status = population.status === 'diverging' ? 'расходится' : population.status === 'fragile' ? 'уязвима' : 'стабильна';
    return `<div class="population-row"><i style="background:${livingBiomeColor(terrain)}"></i><span><b>${safeText(BIOME_NAMES[terrain.biome] || terrain.biome)}</b><small>${status} · изоляция ${population.isolation}</small></span><output>${population.size}</output></div>`;
  }).join('');
}
