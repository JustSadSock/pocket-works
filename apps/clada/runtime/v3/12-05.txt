updateReadouts = function livingUpdateReadouts() {
  livingLegacyUpdateReadouts();
  const populationChip = document.querySelector('#populationCountChip');
  if (populationChip) populationChip.textContent = String(state.populations.filter((population) => population.size > 0).length);
  const seasonChip = document.querySelector('#seasonChip');
  if (seasonChip) seasonChip.textContent = SEASONS[state.season]?.replace('РАННЯЯ ', '').replace('ПОЗДНЯЯ ', '') || '—';
};

function installLivingInterface() {
  const readout = document.querySelector('#liveReadout');
  if (readout && !document.querySelector('#populationCountChip')) {
    readout.insertAdjacentHTML('beforeend', '<span><i class="dot population-group-dot"></i><b id="populationCountChip">0</b> популяций</span><span class="season-readout"><b id="seasonChip">—</b></span>');
  }
  const seeds = document.querySelector('.world-seeds');
  if (seeds && !document.querySelector('.origin-modes')) {
    seeds.insertAdjacentHTML('beforebegin', `
      <div class="origin-modes" aria-label="Возраст биосферы">
        <button type="button" data-origin="mature" aria-pressed="true"><b>ЗРЕЛАЯ БИОСФЕРА</b><small>7–10 видов, пищевая сеть и вымершие предки</small></button>
        <button type="button" data-origin="primordial" aria-pressed="false"><b>ПЕРВИЧНАЯ КОЛОНИЯ</b><small>3 ранних вида и короткая предыстория</small></button>
      </div>`);
  }
  document.querySelector('.origin-modes')?.addEventListener('click', (event) => {
    const button = event.target.closest('[data-origin]'); if (!button) return;
    LIVING.originMode = button.dataset.origin;
    document.querySelectorAll('[data-origin]').forEach((candidate) => candidate.setAttribute('aria-pressed', String(candidate === button)));
    tone(button.dataset.origin === 'mature' ? 370 : 250, .05, .014, 'triangle');
  });
  const introParagraph = document.querySelector('.intro-copy p');
  if (introParagraph) introParagraph.textContent = 'Открой уже сложившуюся биосферу или раннюю колонию. Виды занимают разные ниши, разделяются географией и зависят друг от друга.';
  updateReadouts();
  if (!intro.hidden && state?.generation === 0) removeStoredWorld();
}

installLivingInterface();
