
      population.size = 0;
      if (state.generation - population.lastSeen > 3) population.status = 'extinct';
    }
  }
  state.populations = state.populations.filter((population) => population.status !== 'extinct' || state.generation - population.lastSeen < 18).slice(-120);
}

function livingTopInteraction(map, exclude = null) {
  const entries = Object.entries(map || {}).filter(([key]) => Number(key) !== exclude).sort((a, b) => b[1] - a[1]);
  return entries.length ? Number(entries[0][0]) : null;
}
