onst parentSpecies = state.species.find((species) => species.id === population.speciesId);
  if (!parentSpecies || members.length < 5) return;
  const newId = newSpecies(population.centroid, parentSpecies.id);
  const species = state.species.find((entry) => entry.id === newId);
  species.populationOrigin = population.id;
  species.hybrid = Boolean(population.hybrid);
  species.hybridParentId = population.hybridParentId || null;
  species.habitat = terrainAt(members[0].x, members[0].y).biome;
  species.niche = livingDietLabel(population.centroid);
  population.speciesId = newId;
  population.status = 'species';
  population.isolation = 0;
  for (const organism of members) organism.speciesId = newId;
  state.statistics.speciations += 1;
  state.populationEvents.push({ generation: state.generation, populationId: population.id, type: 'speciation', speciesId: newId });
}

function updatePopulationStructure() {
  const groups = new Map();
  for (const organism of state.organisms) {
    const observedRegion = terrainAt(organism.x, organism.y).region;
    organism.homeRegion ||= observedRegion;
    if (observedRegion !== organism.homeRegion) {
      organism.regionTenure = (organism.regionTenure || 0) + 1;
      if (organism.regionTenure >= 3 && organism.genome.migration > .32) {
        organism.homeRegion = observedRegion;
        organism.regionTenure = 0;
      }
    } else {
      organism.regionTenure = 0;
    }
    const region = organism.homeRegion;
    const key = `${organism.speciesId}:${region}`;
    if (!groups.has(key)) groups.set(key, []);
    groups.get(key).push(organism);
  }
  for (const [key, members] of [...groups.entries()]) {
    if (members.length >= 3) continue;
    const speciesId = key.split(':')[0];
    const alternatives = [...groups.entries()]
      .filter(([otherKey, otherMembers]) => otherKey !== key && otherKey.startsWith(`${speciesId}:`) && otherMembers.length >= 3)
      .sort((left, right) => right[1].length - left[1].length);
    if (!alternatives.length) continue;
    const [targetKey, targetMembers] = alternatives[0];
    const targetRegion = targetKey.split(':')[1];
    for (const organism of members) { organism.homeRegion = targetRegion; organism.regionTenure = 0; }
    targetMembers.push(...members);
    groups.delete(key);
  }
  const seen = new Set();
  for (const [key, members] of groups) {
    const [speciesIdText, region] = key.split(':');
    const speciesId = Number(speciesIdText);
    let population = state.populations.find((entry) => entry.speciesId === speciesId && entry.region === region && entry.status !== 'absorbed');
    if (!population) population = createPopulation(speciesId, region, members[0].genome, 'forming', state.generation);
    seen.add(population.id);
    const previousSize = population.size || 0;
    population.previousSize = previousSize;
    population.size = members.length;
    population.trend = members.length - previousSize;
    population.lastSeen = state.generation;
    population.centroid = livingCentroid(members);
    const parentSpecies = state.species.find((species) => species.id === speciesId);
    population.divergence = parentSpecies ? genomeDistance(population.centroid, parentSpecies.centroid) : 0;
    const otherRegions = [...groups.keys()].some((otherKey) => otherKey.startsWith(`${speciesId}:`) && otherKey !== key);
    population.isolation = otherRegions ? population.isolation + 1 : Math.max(0, population.isolation - 2);
    population.status = population.isolation >= 4 ? 'diverging' : members.length < 4 ? 'fragile' : 'stable';
    for (const organism of members) { organism.populationId = population.id; organism.homeRegion ||= region; }
    const threshold = .17 - state.env.mutation * .035;
    if (population.isolation >= 8 && population.divergence > threshold && members.length >= 5) livingPromotePopulation(population, members);
  }
  for (const population of state.populations) {
    if (!seen.has(population.id) && population.status !== 'absorbed') {
      population.size = 0;
      if (state.generation - population.lastSeen > 3) population.status = 'extinct';
    }
  }
  state.populations = state.populations.filter((population) => population.status !== 'extinct' || state.generation - population.lastSeen < 18).slice(-120);
}

function livingTopInteraction(map, exclude = null) {
  const entries = Object.entries(map || {}).filter(([key]) => Number(key) !== exclude).sort((a, b) => b[1] - a[1]);
  return entries.length ? Number(entries[0][0]) : null;
}
