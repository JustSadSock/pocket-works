function livingPortraitGroup(species, canvasElement) {
  const context = canvasElement.getContext('2d');
  context.fillStyle = '#dfe2cd'; context.fillRect(0, 0, canvasElement.width, canvasElement.height);
  const base = { id: species.id, genome: species.centroid };
  const ancestor = findAncestorSpecies(species);
  if (ancestor) {
    context.save(); context.globalAlpha = .13;
    drawOrganismShape(context, { id: ancestor.id, genome: ancestor.centroid }, 101, 83, 48, 0, true);
    context.restore();
  }
  drawOrganismShape(context, livingDisplayOrganism({ ...base, id: species.id * 3 }, 'f'), 68, 86, 32, -.06, true);
  drawOrganismShape(context, livingDisplayOrganism({ ...base, id: species.id * 3 + 1 }, 'm'), 128, 88, 27, .08, true);
  context.save(); context.globalAlpha = .86; drawOrganismShape(context, livingDisplayOrganism({ ...base, id: species.id * 3 + 2 }, 'f', true), 101, 137, 18, 0, true); context.restore();
}
