function clamp(value, min, max) {
  return Math.max(min, Math.min(max, value));
}

function mulberry32(seed) {
  return function random() {
    let t = seed += 0x6D2B79F5;
    t = Math.imul(t ^ t >>> 15, t | 1);
    t ^= t + Math.imul(t ^ t >>> 7, t | 61);
    return ((t ^ t >>> 14) >>> 0) / 4294967296;
  };
}

function hash2(x, y, seed) {
  let h = Math.imul(x + 374761393, 668265263) ^ Math.imul(y + 1442695041, 2246822519) ^ seed;
  h = Math.imul(h ^ h >>> 13, 1274126177);
  return ((h ^ h >>> 16) >>> 0) / 4294967295;
}

function smoothstep(t) {
  return t * t * (3 - 2 * t);
}

function valueNoise(x, y, seed, scale) {
  const sx = x / scale;
  const sy = y / scale;
  const x0 = Math.floor(sx);
  const y0 = Math.floor(sy);
  const tx = smoothstep(sx - x0);
  const ty = smoothstep(sy - y0);
  const a = hash2(x0, y0, seed);
  const b = hash2(x0 + 1, y0, seed);
  const c = hash2(x0, y0 + 1, seed);
  const d = hash2(x0 + 1, y0 + 1, seed);
  const top = a + (b - a) * tx;
  const bottom = c + (d - c) * tx;
  return top + (bottom - top) * ty;
}

function terrainNoise(x, y, seed) {
  return valueNoise(x, y, seed, 8) * 0.52 + valueNoise(x, y, seed + 91, 4) * 0.31 + valueNoise(x, y, seed + 207, 2) * 0.17;
}

function cellIndex(x, y) {
  return y * COLS + x;
}

function coords(index) {
  return { x: index % COLS, y: Math.floor(index / COLS) };
}

function neighborIndices(index) {
  const { x, y } = coords(index);
  const result = [];
  if (x > 0) result.push(index - 1);
  if (x < COLS - 1) result.push(index + 1);
  if (y > 0) result.push(index - COLS);
  if (y < ROWS - 1) result.push(index + COLS);
  return result;
}

function distanceCells(a, b) {
  const ac = coords(a);
  const bc = coords(b);
  return Math.hypot(ac.x - bc.x, ac.y - bc.y);
}

function makeName(random) {
  return `${PREFIXES[Math.floor(random() * PREFIXES.length)]}${SUFFIXES[Math.floor(random() * SUFFIXES.length)]}`.toUpperCase();
}

function makeVertices(seed) {
  const vertices = [];
  for (let y = 0; y <= ROWS; y += 1) {
    const row = [];
    for (let x = 0; x <= COLS; x += 1) {
      const edgeX = x === 0 || x === COLS;
      const edgeY = y === 0 || y === ROWS;
      const jitterX = edgeX ? 0 : (hash2(x, y, seed + 311) - 0.5) * 0.42;
      const jitterY = edgeY ? 0 : (hash2(x, y, seed + 617) - 0.5) * 0.42;
      row.push({ x: x + jitterX, y: y + jitterY });
    }
    vertices.push(row);
  }
  return vertices;
}
