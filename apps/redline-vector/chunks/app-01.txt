function buildTrack(def) {
  const segments = [];
  let y = 0;
  let index = 0;
  for (const [count, curve, hill] of def.sections) {
    const startY = y;
    const endY = y + hill * 28;
    for (let n = 0; n < count; n += 1) {
      const t = n / Math.max(1, count - 1);
      const nextY = easeInOut(startY, endY, t);
      const palette = def.roadType === 'city' ? (index % 2 ? COLORS.neonDark : COLORS.neonLight) : (index % 2 ? COLORS.dark : COLORS.light);
      segments.push({
        index,
        p1: makePoint(0, y, index * SEGMENT_LENGTH),
        p2: makePoint(0, nextY, (index + 1) * SEGMENT_LENGTH),
        curve: curve * Math.sin(Math.PI * Math.min(1, (n + 1) / count)),
        color: palette,
        tunnel: def.id === 'pass' && ((index > 94 && index < 118) || (index > 210 && index < 229)),
        boostPad: index % 83 === 47,
        checkpoint: false,
        clip: 0
      });
      y = nextY;
      index += 1;
    }
  }
  const checkpointIndexes = [0.25, 0.5, 0.75].map(f => Math.floor(segments.length * f));
  checkpointIndexes.forEach((cp, cpIndex) => { if (segments[cp]) segments[cp].checkpoint = cpIndex + 1; });
  return { ...def, segments, length: segments.length * SEGMENT_LENGTH, checkpointIndexes };
}
