function renderRace(race) {
  const player=race.player;const track=race.track;const base=findSegment(track,player.z);const basePercent=wrap(player.z,track.length)%SEGMENT_LENGTH/SEGMENT_LENGTH;
  const cameraY=lerp(base.p1.world.y,base.p2.world.y,basePercent)+CAMERA_HEIGHT;let cameraX=player.x*ROAD_WIDTH;
  const speedRatio=Math.abs(player.speed)/player.maxSpeed;const shake=profile.settings.cameraShake?state.shake*(Math.random()-.5):0;
  ctx.save();ctx.translate(shake,shake*.5);drawBackground(track,base.curve,speedRatio);
  const drawDistance=profile.settings.lowQuality?115:175;let maxY=H;let x=0;let dx=-(base.curve*basePercent);const baseIndex=base.index;
  const visible=[];
  for(let n=0;n<drawDistance;n+=1){const segment=track.segments[(baseIndex+n)%track.segments.length];segment.looped=segment.index<baseIndex;segment.fog=1-n/drawDistance;segment.clip=maxY;
    const cameraZ=player.z-(segment.looped?track.length:0);segment.p1.world.z=segment.index*SEGMENT_LENGTH+(segment.looped?track.length:0);segment.p2.world.z=(segment.index+1)*SEGMENT_LENGTH+(segment.looped?track.length:0);
    project(segment.p1,cameraX-x,cameraY,cameraZ);project(segment.p2,cameraX-x-dx,cameraY,cameraZ);x+=dx;dx+=segment.curve;
    if(segment.p1.camera.z<=CAMERA_DEPTH||segment.p2.screen.y>=maxY)continue;visible.push(segment);maxY=segment.p2.screen.y;
  }
  for(let i=visible.length-1;i>=0;i-=1){const segment=visible[i];drawRoadSegment(segment,clamp(segment.fog*1.7,0,1));drawTunnel(segment);drawRoadside(track,segment,-1);drawRoadside(track,segment,1);
    const cars=[];for(const ai of race.ai){let dz=wrap(ai.z-player.z,track.length);if(dz<drawDistance*SEGMENT_LENGTH)cars.push({z:ai.z,x:ai.x,car:ai,dist:dz});}
    if(race.mode==='time'&&race.ghost.length){const sample=ghostSample(race);if(sample){const dz=wrap(sample.z-player.z,track.length);if(dz<drawDistance*SEGMENT_LENGTH)cars.push({z:sample.z,x:sample.x,car:{color:'#47d7ff'},dist:dz,ghost:true});}}
    cars.filter(c=>findSegment(track,c.z).index===segment.index).sort((a,b)=>b.dist-a.dist).forEach(c=>drawOpponent(carScreenPosition(race,c.z,c.x),c.car,c.ghost?.48:1,c.ghost));
  }
  drawParticles();drawCockpit(race);ctx.restore();
  if(state.flash>0){ctx.fillStyle=`rgba(255,255,255,${state.flash*.25})`;ctx.fillRect(0,0,W,H);}
}

function ghostSample(race) {
  if (!race.ghost.length || !race.started) return null; const target=race.elapsed; let lo=0,hi=race.ghost.length-1;
  while(lo<hi){const mid=Math.floor((lo+hi)/2);if(race.ghost[mid].t<target)lo=mid+1;else hi=mid;}const b=race.ghost[lo],a=race.ghost[Math.max(0,lo-1)];if(!a||!b)return b;const t=clamp((target-a.t)/Math.max(.001,b.t-a.t),0,1);let bz=b.z;if(bz<a.z)bz+=race.track.length;return{x:lerp(a.x,b.x,t),z:wrap(lerp(a.z,bz,t),race.track.length)};
}

function updateAI(race, dt) {
  const p=race.player;const track=race.track;const adaptive=clamp((p.totalProgress-(race.ai[0]?.totalProgress||0))/track.length,-.08,.08);
  for(const ai of race.ai){if(ai.finished)continue;ai.think-=dt;if(ai.think<=0){ai.think=.25+Math.random()*.45;const seg=findSegment(track,ai.z);let curvePenalty=Math.abs(seg.curve)*(.05+(.5-ai.personality.risk)*.08);let desired=ai.baseSpeed*(1-curvePenalty+adaptive*.25);
      if(ai.personality.id==='inconsistent'&&Math.random()<.16)ai.mistake=.5+Math.random()*1.2;if(ai.mistake>0)desired*=.82;
      let target=(Math.random()-.5)*ai.personality.laneNoise*2;let nearest=null,nearestDz=Infinity;
      const candidates=[...race.ai.filter(other=>other!==ai),{...p,id:'player'}];for(const other of candidates){const dz=wrap(other.z-ai.z,track.length);if(dz>0&&dz<900&&dz<nearestDz){nearest=other;nearestDz=dz;}}
      if(nearest){const passSide=nearest.x>0?-1:1;target=clamp(nearest.x+passSide*(.36+ai.personality.risk*.18),-.92,.92);desired*=nearestDz<330?.93:1;}
      const playerDz=wrap(p.z-ai.z,track.length);if(ai.personality.id==='aggressive'&&playerDz<650&&playerDz>0)target=clamp(p.x+(Math.random()-.5)*.12,-.9,.9);
      if(ai.personality.id==='defensive'){const behind=wrap(ai.z-p.z,track.length);if(behind<700)target=clamp(p.x,-.9,.9);}
      ai.targetX=target;ai.desiredSpeed=desired;if(ai.boost>.15&&Math.abs(seg.curve)<.28&&Math.random()<.22){ai.boosting=true;ai.boostTime=.45+Math.random()*.55;}
    }
    ai.mistake=Math.max(0,ai.mistake-dt);if(ai.boosting){ai.boostTime-=dt;ai.boost-=dt*.13;if(ai.boostTime<=0||ai.boost<=0)ai.boosting=false;}
    ai.speed+=clamp((ai.desiredSpeed||ai.baseSpeed)-ai.speed,-3400*dt,2600*dt);if(ai.boosting)ai.speed+=2100*dt;
